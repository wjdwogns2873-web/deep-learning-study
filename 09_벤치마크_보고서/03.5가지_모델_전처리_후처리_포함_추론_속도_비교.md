# YOLOv8 전처리 및 후처리 포함 추론 속도 비교 (PyTorch vs ONNX Runtime vs TensorRT)

YOLOv8n 모델의 전처리(Resize, BGR to RGB, Normalization) 후처리(NMS) 과정을 모두 포함한 추론 시간을 벤치마킹한 결과입니다.

제 컴퓨터는 M4 맥미니라서 TensorRT 환경 구축이 안되기 때문에 구글 코랩의 T4 GPU를 사용하였습니다.

동일한 조건에서 1000회 추론을 했을 때 평균, 최소, 최대 지연 시간 및 FPS를 측정했습니다.

순수 추론에서도 C++의 평균 처리 속도가 1.68 ms로 가장 빨랐고, 전처리 후처리를 포함한 처리 속도도 평균 5.64 ms로 압도적으로 빠른 걸 보여줍니다.

순수 추론 때와 마찬가지로 C++과 파이썬의 가장 큰 차이인 GIL(Global Interpreter Lock), 그리고..

파이썬의 인터프리터가 매 라인에 대해 객체를 래핑하고 타입을 변환하고 C++ 레이어로 올려주는 과정의 오버헤드가 C++에서는 없기 때문에, 
C++은 극한의 속도를 보여줍니다.

```python
import time
import numpy as np
import torch
from PIL import Image
from ultralytics import YOLO

# 파이토치는 연산 효율을 높이기 위해 한번 사용한 VRAM 공간을 바로 반납하지 않고 내부의 Caching Allocator에 남겨두었다가 다음에 재사용합니다.
# 벤치마킹을 정밀하게 측정할 때는 이전에 실행했던 다른 작업들의 찌꺼기 메모리를 완전히 배제해야 하기때문에 캐시를 비워줍니다.
torch.cuda.empty_cache()
model = YOLO('yolov8n.pt')

# 테스트할 이미지를 로드합니다. 디스크 I/O 병목을 제외하기 위해 측정 루프 바깥쪽에 배치했습니다.
img_path = './test_images/test.jpg'
img = Image.open(img_path)

if img is None:
    raise FileNotFoundError(f'이미지를 찾을 수 없습니다: {img_path}')

# Warm-up 50회
for _ in range(50):
    _ = model(img, verbose=False)
torch.cuda.synchronize()

# 1000회 측정
latencies = []
for _ in range(1000):
    start = time.time()
    
    _ = model(img, verbose=False)
    torch.cuda.synchronize()
    
    latencies.append((time.time() - start) * 1000)

print(f'Avg Latency: {avg:.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print(f'Avg FPS: {1000.0 / avg:.1f} FPS')
print('=================================================================')
```

<summary><b>2. ONNX Runtime 측정 코드 및 결과</b></summary>

```Python
import time
import cv2
import numpy as np
import onnxruntime as ort

# ONNX 추론 세션 생성
# 1순위로 GPU(CUDA)를 시도합니다. GPU와 드라이버가 잘 갖춰져 있다면 곧바로 GPU 가속으로 동작합니다.
# 2순위로 CPU를 시도합니다. 지금 실행한 컴퓨터가 GPU가 없는 상황일 때 CPU로 동작합니다.
providers = ['CUDAExecutionProvider', 'CPUExecutionProvider']
session = ort.InferenceSession('yolov8n.onnx', providers=providers)
input_name = session.get_inputs()[0].name
output_name = session.get_outputs()[0].name

# 전처리 함수입니다.
# 640 640으로 리사이즈하고, BGR을 RGB로 변환하고,
# HWC를 CHW로 변환하고 0~1의 실수값으로 정규화하고 차원을 추가하여 (1, 3, 640, 640)로 shape을 만듭니다.
def preprocess(img):
    img_resized = cv2.resize(img, (640, 640))
    img_rgb = cv2.cvtColor(img_resized, cv2.COLOR_BGR2RGB)
    tensor = img_rgb.transpose(2, 0, 1).astype(np.float32) / 255.0 # (640, 640, 3) -> (3, 640, 640)
    return np.expand_dims(tensor, axis=0) # (3, 640, 640) -> (1, 3, 640, 640)

# 후처리 함수입니다.
# 배치 개수가 1이므로 outputs은 (1, 84, 8400)이기 때문에 squeeze로 차원을 제거하여 (84, 8400)으로 만듭니다.
# 84개의 행 중에서 앞 4개는 xmin, ymin, xmax, ymax.. box 정보이므로, 앞 4개만 가져옵니다.
# 그리고 그 뒤 나머지 80개는 각 클래스에 대한 확신도(score)이므로, 80개를 가져온 뒤에, 행(axis=0)[세로방향]을 기준으로
# 8400개의 박스 각각에 대하여 가장 확신도가 높은 score를 채택을 하게 되면, shape은 (8400,)이 됩니다.
# 그리고 8400개의 score가 확신도 임계값(conf_thresh)를 넘어서면 true, 그렇지 않으면 false이므로,
# valid_mask 역시 score와 같은 shape을 가집니다.
# 최종적으로 이 valid_mask를 사용해서,
# 박스들(boxes) 중에 임계값을 넘어선 box 정보(xmin, ymin, xmax, ymax)만 가져오고,
# 점수들(scores) 중에 임계값을 넘어선 점수만 가져옵니다.
def postprocess(outputs, conf_thresh=0.25):
    preds = np.squeeze(outputs[0]) # (84, 8400)
    boxes = preds[:4, :] # (4, 8400)
    scores = np.max(preds[4:, :], axis=0) # (8400,)
    valid_mask = scores > conf_thresh # (8400,)
    return boxes[:, valid_mask], scores[valid_mask] # (4, a), (a,)

# 테스트할 이미지를 로드합니다. 디스크 I/O 병목을 제외하기 위해 측정 루프 바깥쪽에 두었습니다.
img_path = './test_images/test.jpg'
img = cv2.imread(img_path)

# Warm-up
for _ in range(50):
    input = preprocess(img)
    output = session.run([output_name], {input_name: input})
    _ = postprocess(output)

# 1000회 측정
latencies = []
for _ in range(1000):
    start = time.time()
    
    input = preprocess(img)
    output = session.run([output_name], {input_name: input})
    _ = postprocess(output)
    
    latencies.append((time.time() - start) * 1000)

avg = np.mean(latencies)

print('================ [2. ONNX Runtime End-to-End 1000회] ================')
print(f'Avg Latency: {avg:.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print(f'Avg FPS: {1000.0 / avg:.1f} FPS')
print('======================================================================')
```

```text
Min Latency: 10.92 ms
```

<summary><b>3. Python TensorRT (기본 메모리 할당) 측정 코드 및 결과</b></summary>

```Python
import time
import cv2
import numpy as np
import tensorrt as trt
import pycuda.driver as cuda

# Logger를 생성합니다.
logger = trt.Logger(trt.Logger.WARNING)

# 엔진을 로드합니다.
with open("yolov8n.engine", 'rb') as f, trt.Runtime(logger) as runtime:
    engine = runtime.deserialize_cuda_engine(f.read())

context = engine.create_execution_context()

input_shape = (1, 3, 640, 640)
input_size = trt.volume(input_shape) * np.dtype(np.float32).itemsize
output_shape = tuple(engine.get_tensor_shape(engine.get_tensor_name(1)))
output_size = trt.volume(output_shape) * np.dtype(np.float32).itemsize

# input_size, output_size만큼 메모리를 동적 할당합니다.
d_input = cuda.mem_alloc(input_size)
d_output = cuda.mem_alloc(output_size)

# CPU에서 GPU로 작업을 던질 통로(Stream)를 생성합니다.
stream = cuda.Stream()

# Host(CPU)에서 출력결과를 받아낼 그릇. output_shape 크기의 넘파이 배열을 만듭니다.
host_output = np.empty(output_shape, dtype=np.float32)

context.set_tensor_address(engine.get_tensor_name(0), int(d_input))
context.set_tensor_address(engine.get_tensor_name(1), int(d_output))

def preprocess(img):
    img_resized = cv2.resize(img, (640, 640))
    img_rgb = cv2.cvtColor(img_resized, cv2.COLOR_BGR2RGB)
    tensor = img_rgb.transpose(2, 0, 1).astype(np.float32) / 255.0

    # np.ascontiguousarray(): GPU 메모리에 데이터를 한 번에 복사하기 위해, 데이터를 메모리상에 빈틈없이 한줄로 나란히 정렬.
    # 이렇게 하는 이유?
    # transpose를 할 때 넘파이는 계산 속도를 아끼기 위해 실제 메모리에 있는 데이터의 위치를 뒤바꾸지 않습니다.
    # 겉보기(shape)만 바꾸고, 데이터를 읽는 순서의 규칙만 수정합니다.
    # 여전히 원래 이미지의 순서인 [R1, G1, B1, R2, G2, B2...]로 되어있습니다.
    # 하지만, [R1, R2...], [G1, G2...], [B1, B2...]인 것처럼 속여서 보여줍니다.
    # 이렇게 실제 데이터가 주소대로 나란히 있지 않고 듬성듬성 있는 상태를 Non-contiguous(비연속성) 상태라고 합니다.
    # 이 상태 그대로 GPU로 전송(memcpy_htod)하려고 하면 문제가 생깁니다.
    # GPU로 데이터를 쏘아 보내는 하드웨어 가속 장치는 "메모리 특정 주소부터 특정 주소까지 빈틈없이 이어져 있는 연속적인 덩어리"만
    # 한 번에 초고속으로 복사 가능합니다. 징검다리 상태라면 속도가 느리거나 에러가 납니다.
    # 이를 막기 위해 눈속임을 하지 않고 실제 물리 메모리를 새로 파서 데이터를 순서대로 예쁘게 정렬해줍니다.
    # 이렇게 함으로써 GPU로 초고속 전송이 가능해집니다.
    return np.ascontiguousarray(np.expand_dims(tensor, axis=0))

# 전처리 함수입니다.
# boxes: preds는 (1, 84, 8400)의 shape을 가지기 때문에, preds[0, :4, :]를 함으로써 박스 정보(xmin ~ ymax)를 가져옵니다.
# scores: preds[0, 4:, :]는 (80, 8400)입니다. 여기서 각 네모박스(8400개)의 대한 가장 높은 확신도를 가진 클래스를 가져옵니다.
def postprocess(preds, conf_thresh=0.25):
    boxes = preds[0, :4, :] # (4, 8400)
    scores = np.max(preds[0, 4:, :], axis=0) # (8400,)
    valid_mask = scores > conf_thresh
    return boxes[:, valid_mask], scores[valid_mask]

# 테스트할 이미지를 로드합니다. 디스크 I/O 병목을 제외해야 정확한 추론 시간 측정이 가능합니다.
img_path = './test_images/test.jpg'
img = cv2.imread(img_path)

# 예열 (50회)
for _ in range(50):
    input = preprocess(img)
    cuda.memcpy_htod_async(d_input, input, stream)
    context.execute_async_v3(stream_handle=stream.handle)
    cuda.memcpy_dtoh_async(host_output, d_output, stream)
    stream.synchronize()
    _ = postprocess(host_output)

# 1000회 측정
latencies = []
for _ in range(1000):
    start = time.time()
    
    input = preprocess(img)
    cuda.memcpy_htod_async(d_input, input, stream)
    context.execute_async_v3(stream_handle=stream.handle)
    cuda.memcpy_dtoh_async(host_output, d_output, stream)
    stream.synchronize()
    _ = postprocess(host_output)
    
    latencies.append((time.time() - start) * 1000)

avg = np.mean(latencies)

print('================ [3. Python TRT (기본) End-to-End 1000회] ================')
print(f'Avg Latency: {avg:.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print(f'Avg FPS: {1000.0 / avg:.1f} FPS')
print('===========================================================================')
```

<summary><b>4. Python TensorRT (Zero-Copy Pinned Memory) 측정 코드 및 결과</b></summary>

```Python
import time
import cv2
import numpy as np
import tensorrt as trt
import pycuda.driver as cuda

# Logger를 생성합니다.
logger = trt.Logger(trt.Logger.WARNING)

# 엔진을 로드합니다.
with open("yolov8n.engine", 'rb') as f, trt.Runtime(logger) as runtime:
    engine = runtime.deserialize_cuda_engine(f.read())

context = engine.create_execution_context()

input_shape = (1, 3, 640, 640)

# 출력 텐서 shape만 튜플로 감싸주는 이유
# engine.get_tensor_shape()은 tuple이나 list를 반환하지 않고, Dims 클래스 객체를 반환합니다.
# 하지만 cuda.pagelocked_empty()는 NumPy나 PyCUDA 기반이라서 반드시 tuple이나 list로 shape을 받아야 합니다.
output_shape = tuple(engine.get_tensor_shape(engine.get_tensor_name(1)))

# Pinned memory(page-locked), Device Memory 할당
# pagelocked_empty의 의미
# CPU 메모리를 할당할 때 메모리스왑하지 말고 RAM에 딱 고정(Pinned)시켜달라고 요청합니다.
h_input = cuda.pagelocked_empty(input_shape, dtype=np.float32)
h_output = cuda.pagelocked_empty(output_shape, dtype=np.float32)

# h_input, h_output의 크기만큼 메모리를 미리 할당합니다.
d_input = cuda.mem_alloc(h_input.nbytes)
d_output = cuda.mem_alloc(h_output.nbytes)

# CPU에서 GPU로 작업을 던질 통로(Stream)을 생성합니다.
stream = cuda.Stream()

# set_tensor_address(name, ptr)
# 추론을 수행할 때 입력 및 출력 텐서의 GPU 메모리 버퍼 주소(포인터)를 지정하는 메서드입니다.
# 텐서의 이름(name)과 메모리 주소(ptr)를 직접 매핑합니다.
context.set_tensor_address(engine.get_tensor_name(0), int(d_input))
context.set_tensor_address(engine.get_tensor_name(1), int(d_output))

# 고정된 메모리 공간(pinned_buf)에 입력 데이터(tensor)를 복사해서 집어넣습니다.
def preprocess_to_pinned(img, pinned_buf):
    img_resized = cv2.resize(img, (640, 640))
    img_rgb = cv2.cvtColor(img_resized, cv2.COLOR_BGR2RGB)
    tensor = img_rgb.transpose(2, 0, 1).astype(np.float32) / 255.0
    np.copyto(pinned_buf, np.expand_dims(tensor, axis=0))

def postprocess(preds, conf_thresh=0.25):
    boxes = preds[0, :4, :]
    scores = np.max(preds[0, 4:, :], axis=0)
    valid_mask = scores > conf_thresh
    return boxes[:, valid_mask], scores[valid_mask]

img_path = './test_images/test.jpg'
img = cv2.imread(img_path)

# Warm-up
for _ in range(50):
    preprocess_to_pinned(img, h_input)
    cuda.memcpy_htod_async(d_input, h_input, stream)
    context.execute_async_v3(stream_handle=stream.handle)
    cuda.memcpy_dtoh_async(h_output, d_output, stream)
    stream.synchronize()
    _ = postprocess(h_output)

# 1000회 측정
latencies = []
for _ in range(1000):
    start = time.time()
    
    preprocess_to_pinned(img, h_input)
    cuda.memcpy_htod_async(d_input, h_input, stream)
    context.execute_async_v3(stream_handle=stream.handle)
    cuda.memcpy_dtoh_async(h_output, d_output, stream)
    stream.synchronize()
    _ = postprocess(h_output)
    
    latencies.append((time.time() - start) * 1000)

avg = np.mean(latencies)

print('================ [4. Python TRT (Zero-Copy) End-to-End 1000회] ================')
print(f'Avg Latency: {avg:.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print(f'Avg FPS: {1000.0 / avg:.1f} FPS')
print('================================================================================')
```

<summary><b>5. C++ TensorRT 구현 코드 및 측정 결과</b></summary>

### `YoloTRT.h`
```cpp
#ifndef YOLO_TRT_H
#define YOLO_TRT_H

#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <algorithm>
#include <opencv2/opencv.hpp>
#include <NvInfer.h>
#include <cuda_runtime_api.h>

// 탐지된 박스 정보를 표현할 Detection 구조체를 만듭니다.
// xmin, ymin, width, height를 담을 box
// 확신도를 담을 confidence
// 클래스 id를 담을 classId
struct Detection {
    cv::Rect box;
    float confidence;
    int classId;
};

// TensorRT이기 때문에 Logger를 구현합니다.
class TRTLogger : public nvinfer1::ILogger {
    void log(Severity severity, const char* msg) noexcept override {
        if (severity <= Severity::kWARNING) {
            std::cout << "[TensorRT] " << msg << std::endl;
        }
    }
};

class YoloTRT {
public:
    YoloTRT(const std::string& enginePath);
    ~YoloTRT();

    // 실제로 추론할 detect 함수를 선언합니다.
    // 이미지, conf 임계값과 nms 임계값을 매개변수로 받아서 std::vector<Detection>을 반환합니다.
    std::vector<Detection> detect(const cv::Mat& srcImg, float confThresh = 0.25f, float nmsThresh = 0.45f);
    void inferOnly();

private:
    // 전처리 함수입니다. 640 640 리사이즈, HWC -> CHW, BGR to RGB 등의 전처리 작업을 진행합니다.
    void preprocess(const cv::Mat& srcImg);

    // 후처리 함수입니다.
    std::vector<Detection> postprocess(int origW, int origH, float confThresh, float nmsThresh);

private:
    TRTLogger gLogger;
    // nvinfer1이 인터페이스라서 인스턴스를 직접 생상할 수 없기 때문에 오직 포인터로만 다뤄야 합니다.
    nvinfer1::IRuntime* runtime = nullptr;
    nvinfer1::ICudaEngine* engine = nullptr;
    nvinfer1::IExecutionContext* context = nullptr;

    cudaStream_t stream = nullptr;

    // Device(GPU)의 입출력 포인터(주소값)입니다.
    void* d_input = nullptr;
    void* d_output = nullptr;

    // Host(CPU)의 입출력 포인터(주소값)입니다.
    float* h_input = nullptr;
    float* h_output = nullptr;

    size_t inputSize = 1 * 3 * 640 * 640 * sizeof(float);
    size_t outputSize = 1 * 84 * 8400 * sizeof(float);

    std::vector<cv::Rect> boxes; // 탐지된 네모박스를 담기 위해 std::vector<cv::Rect>를 선언합니다.
    std::vector<float> confidences; // 네모박스들의 확신도를 담기 위해 std::vector<float>을 선언합니다.
    std::vector<int> classIds; // 네모박스들의 클래스 id를 담기 위해 std::vector<int>를 선언합니다.
    std::vector<int> indices; // conf, nms 임계값을 넘은 네모박스의 인덱스를 담을 std::vector<int>를 선언합니다.
};

#endif // YOLO_TRT_H
```

### `YoloTRT.cpp`
```cpp
#include "YoloTRT.h"

YoloTRT::YoloTRT(const std::string& enginePath) {
    std::ifstream file(enginePath, std::ios::binary);
    if (!file.good()) {
        std::cerr << "엔진 파일을 찾을 수 없습니다: " << enginePath << std::endl;
        return; 
    }
    file.seekg(0, file.end); // file 커서를 맨 끝(end)으로 이동
    size_t size = file.tellg(); // 현재 커서의 위치(end)는 곧 파일의 전체 바이트 크기이므로, 이것을 size_t 변수에 저장합니다.
    file.seekg(0, file.beg); // file 커서를 다시 맨 처음(beg)으로 이동

    std::vector<char> engineData(size); // file 크기만큼의 빈 공간을 가진 1차원 벡터 생성
    // 벡터가 메모리에 할당된 시작 주소(engineData.data())를 통해,
    // file에 들어있던 바이너리 데이터를 그 시작 주소로부터 size만큼 통째로 읽어와서 복사합니다.
    file.read(engineData.data(), size);
    file.close();

    runtime = nvinfer1::createInferRuntime(gLogger);

    // engineData에는 .engine 파일의 바이너리가 1차원 벡터로 담겨있는 상태인데, .data()를 써서 그 1차원 벡터의 주소 시작값을 넘겨주고,
    // 주소 시작값부터 두 번째 인자(size)만큼의 데이터를 역직렬화해서 엔진을 생성합니다.
    engine = runtime->deserializeCudaEngine(engineData.data(), size);
    context = engine->createExecutionContext();

    // CPU에서 GPU로 데이터를 전송할 통로(Stream) 생성
    cudaStreamCreate(&stream);

    // inputSize, outputSize만큼의 메모리를 미리 동적 할당합니다.
    // 그리고 그 할당된 메모리의 시작 주소를 d_input, d_output에 담아줍니다.
    // 여기서 참조(&)로 던져주는 이유..
    // 그냥 d_input으로 던져주게 되면 그냥 값만 넘겨주는거지, 주소를 넘겨주진 못하기 때문에 d_input에는 변화가 없다.
    // 그래서 &d_input으로 넘겨줘야, 문신처럼 각인할 수 있다.
    cudaMalloc(&d_input, inputSize);
    cudaMalloc(&d_output, outputSize);

    // cudaHostAlloc(): CUDA에서 Pinned Memory를 Host(CPU) 측에 할당하는 함수입니다.
    // 일반적으로 malloc()을 통해 할당한 메모리는 OS에 의해 디스크로 페이징(메모리스왑)이 될 수 있는 Pageable Memory입니다.
    // cudaHostAlloc()으로 할당된 메모리는 물리 RAM에 상주할 수 있으므로, GPU가 Host Memory에 빠르게 접근할 수 있습니다.
    // &h_input으로 넘겨주는 이유..
    // 그냥 h_input으로 넘겨주면 함수 내부에서는 nullptr이라는 값만 복사해가고, h_input은 변화가 없다.(계속 nullptr로 남음)
    // 그래서 포인터를 참조할 수 있게 &를 붙여줍니다.
    // (void**): CUDA 함수와 타입을 맞추기 위한 형변환
    // cudaHostAlloc 함수는 사용자가 메모리에 float을 담을지, int를 담을지 알 수 없기 때문에 어떤 포인터 주소든 다 받아줄 수 있는 void** 타입으로 합니다.
    cudaHostAlloc((void**)&h_input, inputSize, cudaHostAllocDefault);
    cudaHostAlloc((void**)&h_output, outputSize, cudaHostAllocDefault);

    // 각 벡터 안에는 최대 8400개의 데이터가 들어올 수 있으므로 미리 메모리 공간을 예약합니다.
    boxes.reserve(8400);
    confidences.reserve(8400);
    classIds.reserve(8400);
    indices.reserve(8400);

    const char* inputName = engine->getIOTensorName(0);
    const char* outputName = engine->getIOTensorName(1);
    // 모델의 입력과 출력 데이터가 실제로 저장될 메모리 주소(d_input, d_output)를 매핑해줍니다.
    context->setTensorAddress(inputName, d_input);
    context->setTensorAddress(outputName, d_output);
}

YoloTRT::~YoloTRT() {
    // 추론이 끝나면 모든 자원을 반납합니다.
    if (stream) cudaStreamDestroy(stream);
    if (d_input) cudaFree(d_input);
    if (d_output) cudaFree(d_output);

    if (h_input) cudaFreeHost(h_input);
    if (h_output) cudaFreeHost(h_output);

    if (context) delete context;
    if (engine) delete engine;
    if (runtime) delete runtime;
}

// 전처리 함수힙니다.
void YoloTRT::preprocess(const cv::Mat& srcImg) {
    cv::Mat resizedImg;
    cv::resize(srcImg, resizedImg, cv::Size(640, 640)); // 640 640으로 리사이즈합니다.
    cv::cvtColor(resizedImg, resizedImg, cv::COLOR_BGR2RGB); // BGR을 RGB 순서로 변환합니다.

    cv::Mat floatImg;
    resizedImg.convertTo(floatImg, CV_32FC3, 1.0 / 255.0); // 0~1의 실수값으로 값을 정규화합니다.

    // reinterpret_cast: 이 메모리 주소의 해석 방식을 강제로 바꿔라는 뜻입니다.
    // cv::Mat 객체(floatImg)의 내부 데이터의 실제 주소는 uchar* 형태입니다.
    // 이 메모리 주소를 float* 형식을 가진 포인터로 취급해서, 앞으로 실수 단위로 값을 꺼내 쓸 수 있도록 해달라고 타입 변환하는 것입니다.
    int channelSize = 640 * 640;
    float* data = reinterpret_cast<float*>(floatImg.data);

    // [R0, G0, B0, R1, G1, B1, ...]를 [R0, R1, ..., G0, G1, ..., B0, B1, ...]로 바꿉니다.
    // 그러면 HWC가 CHW로 바뀝니다. (640, 640, 3) -> (3, 640, 640)
    for (int i = 0; i < channelSize; ++i) {
        h_input[i]                   = data[i * 3 + 0]; // R
        h_input[i + channelSize]     = data[i * 3 + 1]; // G
        h_input[i + channelSize * 2] = data[i * 3 + 2]; // B
    }
}

// 후처리 함수입니다.
std::vector<Detection> YoloTRT::postprocess(int origW, int origH, float confThresh, float nmsThresh) {
    // 이전 프레임 때 사용했던 네 개의 벡터들을 깔끔하게 비워줍니다.
    boxes.clear();
    confidences.clear();
    classIds.clear();
    indices.clear();

    // 전처리 과정에서 원본 이미지를 640 640으로 리사이즈 했으므로, 탐지된 박스도 640 640 기준일 것입니다.
    // 이것을 원본으로 복원하는 과정이 필요하기 때문에, 
    // 원본 이미지의 가로(origW), 세로(origH) 길이를 각각 640으로 나눈 뒤에 scaleX, scaleY로 저장합니다.
    // 나중에 이 값을 곱해주면 됩니다.
    float scaleX = static_cast<float>(origW) / 640.0f;
    float scaleY = static_cast<float>(origH) / 640.0f;

    // 탐지된 8400개의 박스에 대해 루프를 돕니다.
    for (int i = 0; i < 8400; i++) {
        // h_output은 (1, 84, 8400)의 shape을 가지고 있습니다.
        // 앞 4개(0~3)는 cx, cy, w, h를 가지고 있습니다.
        float cx = h_output[0 * 8400 + i];
        float cy = h_output[1 * 8400 + i];
        float w  = h_output[2 * 8400 + i];
        float h  = h_output[3 * 8400 + i];

        // 각 80개의 클래스 중에서 확신도가 가장 높은 클래스 아이디와, 그 클래스의 확신도를 가져오는 로직입니다.
        float maxScore = 0.0f;
        int maxClassId = -1;
        // 뒤 80개(4~83)는 클래스 아이디를 의미합니다.
        for (int c = 0; c < 80; c++) {
            float score = h_output[(c + 4) * 8400 + i];
            if (score > maxScore) {
                maxScore = score;
                maxClassId = c;
            }
        }

        // 가장 높은 확신도를 가진 클래스의 확신도가 임계값을 넘어갔을 때의 박스(그리고 확신도, 클래스 아이디)만 벡터에 담습니다.
        if (maxScore > confThresh) {
            float x1 = (cx - w / 2.0f) * scaleX;
            float y1 = (cy - h / 2.0f) * scaleY;
            float boxW = w * scaleX;
            float boxH = h * scaleY;

            cv::Rect box(x1, y1, boxW, boxH); // 탐지된 네모박스입니다.
            cv::Rect bounds(0, 0, origW, origH); // 원본 이미지입니다.
            // 탐지된 네모박스 구했다면서? 근데 왜 원본 이미지와 교집합을 또 구하는데?
            // YOLOv8 모델이 탐지한 네모박스 좌표가 원본 이미지를 벗어났을 수도 있다.
            // 예를 들어, 원본 이미지가 640 640이라고 가정하고, 탐지된 물체는 x1 값이 480이고, boxW가 170이라고 했을 때,
            // xmin은 480, xmax는 650(480 + 170)이 되어버리면 640을 초과하게 된다.
            // 이렇게 되면 오류가 발생할 수 있기 때문에, 원본 이미지와의 교집합을 구함으로써 보정하는 과정이 필요하다.
            box &= bounds; // 원본 이미지와 탐지된 네모박스의 교집합을 구합니다.

            // 탐지된 네모박스, 그 네모박스의 클래스 아이디, 그 클래스에 대한 확신도를 각각의 벡터에 담습니다.
            boxes.emplace_back(box);
            confidences.emplace_back(maxScore);
            classIds.emplace_back(maxClassId);
        }
    }

    // 탐지된 박스와 확신도를 담은 벡터, 그리고 확신도 임계값, NMS 임계값을 통해 총 8400개의 박스 중에
    // 임계값을 만족하는 박스의 인덱스가 몇번인지를 추출하는 함수입니다.
    // indices에 인덱스 번호가 담깁니다. (예: 54, 678, 8300)
    cv::dnn::NMSBoxes(boxes, confidences, confThresh, nmsThresh, indices);

    // 추출된 인덱스 번호(indices)만을 가지고 Detection 구조체를 담는 벡터(detections)에 하나씩 밀어넣습니다.
    std::vector<Detection> detections;
    detections.reserve(indices.size());
    for (int idx : indices) {
        detections.push_back({ boxes[idx], confidences[idx], classIds[idx] });
    }
    // 최종적으로 그 벡터를 반환합니다.
    return detections;
}

// 실제 추론 함수입니다.
std::vector<Detection> YoloTRT::detect(const cv::Mat& srcImg, float confThresh, float nmsThresh) {
    // 이미지(프레임)가 비어 있는지 확인합니다.
    if (srcImg.empty()) return {};
    // 전처리 과정을 거칩니다.
    preprocess(srcImg);

    // h_input(Host, CPU)에는 이미지의 RGB 정보가 가지런히 담겨 있고, 그것을 stream을 통해 d_input(Device, GPU)으로 복사해줍니다. inputSize만큼.
    cudaMemcpyAsync(d_input, h_input, inputSize, cudaMemcpyHostToDevice, stream);
    context->enqueueV3(stream);
    // 추론을 끝낸 d_output(Device, GPU)에는 (1, 84, 8400)의 데이터가 메모리 공간에 가지런히 담겨 있고, 그것을 stream을 통해 h_output(Host, CPU)으로 복사해줍니다.
    cudaMemcpyAsync(h_output, d_output, outputSize, cudaMemcpyDeviceToHost, stream);
    // 처리가 끝날 때까지 기다립니다. 동기화.
    cudaStreamSynchronize(stream);
    // 후처리 과정을 진행합니다.
    return postprocess(srcImg.cols, srcImg.rows, confThresh, nmsThresh);
}

void YoloTRT::inferOnly() {
    context->enqueueV3(stream);
    cudaStreamSynchronize(stream);
}
```

### `main.cpp`
```cpp
#include "YoloTRT.h"
#include <chrono>
#include <numeric>
#include <vector>
#include <iostream>
#include <algorithm>

int main() {
    // YoloTRT 클래스 객체 생성. 엔진 파일의 경로를 매개변수로 던져줍니다.
    // 이렇게 하면 생성자가 호출되며, 엔진 로드 및 입출력 데이터 메모리 동적 할당 등의 준비 작업이 끝납니다.
    YoloTRT detector("yolov8n.engine");

    // 추론 테스트할 이미지를 준비합니다.
    std::string imgPath = "./test_images/test.jpg";
    cv::Mat img = cv::imread(imgPath);

    if (img.empty()) {
        std::cerr << "이미지를 읽을 수 없습니다: " << imgPath << std::endl;
        return -1;
    }

    // 예열 (50회)
    for (int i = 0; i < 50; ++i) {
        detector.detect(img, 0.25f, 0.45f);
    }

    const int iterations = 1000;
    std::vector<double> latencies;
    latencies.reserve(iterations);

    // 1000회 측정
    for (int i = 0; i < iterations; ++i) {
        auto start = std::chrono::high_resolution_clock::now();

        auto detections = detector.detect(img, 0.25f, 0.45f);

        auto end = std::chrono::high_resolution_clock::now();
        double ms = std::chrono::duration<double, std::milli>(end - start).count();
        latencies.push_back(ms);
    }

    double sum = std::accumulate(latencies.begin(), latencies.end(), 0.0);
    double avg = sum / iterations;

    // min과 max를 구하는 방식이 조금 독특합니다. 별(*)을 붙이는 이유
    // std::min_element가 최솟값 숫자 자체를 반환하는 게 아니라, 최솟값이 위치한 주소값을 반환하기 위함입니다.
    // 별(*)을 붙여 역참조를 해서 주소에 있는 알맹이 값을 가져옵니다.
    double min = *std::min_element(latencies.begin(), latencies.end());
    double max = *std::max_element(latencies.begin(), latencies.end());

    std::cout << "================ [5. C++ Native TRT End-to-End 1000회] ================" << std::endl;
    std::cout << "평균 Latency (Avg): " << cv::format("%.2f", avg) << " ms" << std::endl;
    std::cout << "최소 Latency (Min): " << cv::format("%.2f", min) << " ms" << std::endl;
    std::cout << "최대 Latency (Max): " << cv::format("%.2f", max) << " ms" << std::endl;
    std::cout << "평균 FPS: " << cv::format("%.1f", 1000.0 / avg) << " FPS" << std::endl;
    std::cout << "=========================================================================" << std::endl;

    return 0;
}
```
