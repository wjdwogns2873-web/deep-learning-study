# YOLOv8 환경별 추론 속도 비교 (PyTorch, ONNX Runtime, TensorRT)

YOLOv8n 모델의 전처리 후처리 과정을 제외하고 순수 추론 시간을 벤치마킹한 결과입니다.

제 컴퓨터는 M4 맥미니라서 TensorRT 환경 구축이 안되기 때문에 구글 코랩의 T4 GPU를 사용하였습니다.

동일한 조건에서 1000회 추론을 했을 때 평균, 최소, 최대 지연 시간을 측정했습니다.

C++ TensorRT가 압도적으로 빠른 속도를 기록한 이유를 분석해 보았습니다.

1. 생성자 함수를 호출할 때, `cudaMalloc`으로 입력 텐서, 출력 텐서 VRAM 공간을 미리 고정하여 확보하고 재사용함으로써, 매 추론 시 발생하는 메모리 동적 할당 및 해제 오버헤드가 없어졌습니다.

2. 파이썬은 GIL(Global Interpreter Lock)으로 인해, 병렬 처리가 안되기도 하고 매번 파이썬 인터프리터가 매 라인을 객체 래핑하고 타입 변환하고 C++ 레이어로 올려주는 과정에서 오버헤드가 발생하는데, C++은 그러한 오버헤드 없이 바인딩을 통해 TensorRT 엔진 컨텍스트(`IExecutionContext`)를 직접 제어(`enqueueV3`)하여 순수 GPU 연산 효율을 극한으로 끌어올렸습니다.

<summary><b>1. PyTorch (.pt) 측정 코드 및 결과</b></summary>

```python
import time
import numpy as np
import torch
from ultralytics import YOLO

pt_model = YOLO('yolov8n.pt')

# 파이토치는 기본적으로 모델을 로드하면 학습과 추론을 모두 지원하기 때문에, 추론만 할 때는 불필요한 기능들을 꺼줘야 합니다.
# .eval(): 배치정규화(BatchNorm)나 Dropout같은 학습할 때 쓰는 기능들은 전부 OFF 시킨다는 의미입니다.
# .cuda(): 모델의 가중치 데이터들을 CPU에서 GPU VRAM으로 이동시킵니다. 이것을 하지 않으면 GPU로 추론을 할 수 없습니다.

# 여기서 pt_model은 High level Wrapper 객체이고, 그 뒤에 붙은 .model은 내부에서 실제로 돌아가는 순수 파이토치 신경망 객체(nn.Module)입니다.
# 이 순수 파이토치 신경망을 직접 다루었기 때문에, dropout 배치정규화를 끄기 위해 .eval()을 호출해주어야 하고,
# 불필요한 역전파를 하지 않기 위해 torch.no_grad()를 했습니다.
model = pt_model.model.eval().cuda()

dummy_input = torch.ones((1, 3, 640, 640), dtype=torch.float32).cuda()

# 예열 (Warm-up) 50회
with torch.no_grad():
    for _ in range(50):
        _ = model(dummy_input)
torch.cuda.synchronize()

latencies = []

# 파이토치의 자동 미분 엔진을 일시 정지시킵니다.
# 역전파를 위한 연산 그래프를 그리지 않기 때문에 메모리를 아끼고 순수 추론 속도를 올릴 수 있습니다.
# 순수 추론만 진행할거기 때문에 오차를 계산할 필요가 없으니 역전파도 필요없는겁니다.
with torch.no_grad():
    for _ in range(1000):
        start = time.time()

        _ = model(dummy_input)
	# GPU 연산이 끝날 때까지 대기
	# 이것을 안해주면 어떻게 되나요??
	# 파이썬에서 GPU로 모델 연산을 시킬때 GPU 연산의 비동기 특성 때문에 CPU는 추론 결과를 기다리지 않고 바로 다음 코드로 넘어갑니다.
	# 그래서 모델 추론이 1초가 걸리든 10초가 걸리든 측정 시간은 0.00x초가 나올 수 있습니다.
	# 이렇게, GPU 연산이 끝날 때까지 대기해야, 지금 연산의 real latency를 구할 수 있습니다.
        torch.cuda.synchronize()

        latencies.append((time.time() - start) * 1000)

print('================ [1. PyTorch 1000회 벤치마크] ================')
print(f'Avg Latency: {np.mean(latencies):.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print('==============================================================')
```

<summary><b>2. ONNX Runtime 측정 코드 및 결과</b></summary>

```python
import time
import numpy as np
import onnxruntime as ort

# ONNX 추론 세션 생성
# 1순위로 GPU(CUDA)를 시도합니다. T4 같은 GPU와 드라이버가 잘 갖추어져 있다면 곧바로 GPU 가속으로 동작합니다.
# 2순위로 CPU를 시도합니다. 지금 실행한 컴퓨터가 GPU가 없는 상황일 때 CPU로 동작합니다.
providers = ['CUDAExecutionProvider', 'CPUExecutionProvider']
session = ort.InferenceSession('yolov8n.onnx', providers=providers)

# ONNX 모델은 내부적으로 입력/출력 노드의 이름(name)을 기반으로 데이터를 주고받기 때문에, 이름 추출이 필요합니다.
input_name = session.get_inputs()[0].name
output_name = session.get_outputs()[0].name

dummy_input = np.ones((1, 3, 640, 640), dtype=np.float32)

# 예열 (Warm-up)
for _ in range(50):
    _ = session.run([output_name], {input_name: dummy_input})

latencies = []
for _ in range(1000):
    start = time.time()

    # 실제 추론 실행합니다.
    # 지정한 출력 이름(output_name)으로 결과 텐서를 받아옵니다.
    # 입력은 딕셔너리 형태로 입력 이름(input_name)과 텐서 데이터(dummy_input)를 매핑하여 모델에게 전달합니다.
    _ = session.run([output_name], {input_name: dummy_input})

    latencies.append((time.time() - start) * 1000)

print('================ [2. ONNX Runtime 1000회 벤치마크] ================')
print(f'Avg Latency: {np.mean(latencies):.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print('==============================================================')
```

<summary><b>TensorRT 가속 엔진 (.engine) 변환 코드</b></summary>

```python
import os
import tensorrt as trt

# Logger와 Builder를 생성합니다.
# Logger: 모델 빌드 중에 에러가 발생하면 그 에러를 수집하고 출력하는 역할을 합니다.
# Builder: 이 모델을 GPU에 맞게 최적화된 상태로 구워 달라고 명령을 내리는 빌더입니다.
TRT_LOGGER = trt.Logger(trt.Logger.WARNING)
builder = trt.Builder(TRT_LOGGER)

# network와 OnnxParser를 생성합니다.
# network: 모델의 레이어와 구조를 담을 빈 설계도입니다.
# parser: ONNX 파일을 TensorRT가 이해할 수 있도록 변환해서 설계도(network)에 그려주는 역할을 합니다.
network = builder.create_network()
parser = trt.OnnxParser(network, TRT_LOGGER)

# ONNX 파일 파싱
onnx_path = 'yolov8n.onnx'
if not os.path.exists(onnx_path):
    raise FileNotFoundError(f'{onnx_path} 파일이 없습니다.')

with open(onnx_path, 'rb') as model:
    if not parser.parse(model.read()):
	print('ONNX 파싱 실패')
	raise RuntimeError('ONNX 파서 에러')

# config는 이 모델을 굽는데 필요한 설정(최적화를 위한 옵션)입니다.
config = builder.create_builder_config()

# config를 바탕으로 설계도(network)를 구워서 직렬화된 바이트 데이터로 만듭니다.
serialized_engine = builder.build_serialized_network(network, config)

if serialized_engine is None:
    raise RuntimeError('가속 엔진 빌드 프로세스 실패')

# .engine 파일로 저장합니다.
engine_path = 'yolov8n.engine'
with open(engine_path, 'wb') as f:
    f.write(serialized_engine)

# 엔진을 로드하고, 추론을 실행할 context를 준비합니다.
runtime = trt.Runtime(TRT_LOGGER)
with open(engine_path, 'rb') as f:
    engine = runtime.deserialize_cuda_engine(f.read())

context = engine.create_execution_context()
```

<summary><b>3. Python TensorRT (기본 메모리 할당) 측정 코드 및 결과</b></summary>

```python
import time
import numpy as np
import tensorrt as trt
import pycuda.driver as cuda

# 입출력 텐서의 Shape과 전체 바이트 크기 계산 하고 나서 GPU VRAM에 메모리를 미리 할당합니다.
input_shape = (1, 3, 640, 640)
input_size = trt.volume(input_shape) * np.dtype(np.float32).itemsize

output_shape = engine.get_tensor_shape(engine.get_tensor_name(1)) # (1, 84, 8400)
output_size = trt.volume(output_shape) * np.dtype(np.float32).itemsize

d_input = cuda.mem_alloc(input_size)
d_output = cuda.mem_alloc(output_size)

# GPU에게 작업을 던질 통로(Stream)를 생성합니다.
stream = cuda.Stream()

# Host(CPU)에서 사용할 더미 입출력 데이터를 생성합니다.
host_input = np.ones(input_shape, dtype=np.float32)
host_output = np.empty(output_shape, dtype=np.float32)

# 모델의 입력과 출력 데이터가 실제로 저장될 메모리 주소를 TensorRT 엔진에게 직접 지정해 주는 작업입니다.
context.set_tensor_address(engine.get_tensor_name(0), int(d_input))
context.set_tensor_address(engine.get_tensor_name(1), int(d_output))

# 예열 (50회)
for _ in range(50):
    # Host(CPU) 메모리의 입력 데이터(host_input)를 통로(stream)를 통해 Device(GPU) 입력 버퍼(d_input)로 비동기(async)로 전송합니다.
    cuda.memcpy_htod_async(d_input, host_input, stream)
    # TensorRT 비동기 추론 실행합니다.
    context.execute_async_v3(stream_handle=stream.handle)
    # 연산이 끝난 Device(Host) 출력 데이터(d_output)를 Host(CPU) 메모리의 데이터(host_output) 비동기(async)로 회수합니다.
    cuda.memcpy_dtoh_async(host_output, d_output, stream)

    # 정확한 벤치마킹을 위해 Stream 내 연산이 완료될 때까지 대기합니다. 동기화 과정
    stream.synchronize()

# 1000회 추론 측정
latencies = []
for _ in range(1000):
    start = time.time()

    # 이미 바인딩된 GPU 메모리를 바탕으로 순수 추론을 실행합니다.
    context.execute_async_v3(stream_handle=stream.handle)
    stream.synchronize()
    
    latencies.append((time.time() - start) * 1000)

print('================ [3. Python TensorRT (기본) 1000회 벤치마크] ================')
print(f'Avg Latency: {np.mean(latencies):.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print('=========================================================================')
```

```text
Avg Latency: 3.36 ms
Min Latency: 3.24 ms
Max Latency: 5.89 ms
```

<summary><b>4. Python TensorRT (Zero-Copy Pinned Memory) 측정 코드 및 결과</b></summary>

```python
import time
import numpy as np
import tensorrt as trt
import pycuda.driver as cuda

input_shape = (1, 3, 640, 640)

# 출력 텐서 shape만 튜플(tuple)로 감싸주는 이유
# 최신 TensorRT에서 engine.get_tensor_shape()은 tuple이나 list를 반환하지 않고, Dims 클래스 객체를 반환합니다.
# 하지만 cuda.pagelocked_empty()는 NumPy나 PyCUDA 기반이라서 반드시 tuple이나 list 형태의 shape을 받아야 합니다.
output_shape = tuple(engine.get_tensor_shape(engine.get_tensor_name(1)))

# Pinned Memory(Page-locked), Device Memory 할당
# pagelocked_empty의 의미
# CPU 메모리를 할당할 때, "이 메모리는 절대 디스크로 페이지 아웃시키지 말고, RAM에 딱 고정(Pinned)시켜줘"라고 요청합니다.
# 이를 고정 메모리(Pinned Memory)라고도 하는데, 물리적 RAM이 부족할 때 SSD의 일부 공간을 메모리처럼 빌려 쓰는 메모리 스왑을 금지시킵니다.
# 왜냐하면 SSD는 RAM보다 읽기/쓰기 속도가 훨씬 느리기 때문입니다.
h_input = cuda.pagelocked_empty(input_shape, dtype=np.float32)
h_output = cuda.pagelocked_empty(output_shape, dtype=np.float32)

# h_input, h_output의 크기(nbytes)만큼 메모리를 미리 할당합니다.
d_input = cuda.mem_alloc(h_input.nbytes)
d_output = cuda.mem_alloc(h_output.nbytes)

# GPU에게 작업을 던질 통로(Stream)를 생성합니다.
stream = cuda.Stream()

# 넘파이 배열 데이터(여기서는 np.ones)를 고정(Pinned)해둔 메모리 공간(h_input)으로 복사해서 집어넣는 함수입니다.
# 쓰는 이유?
# 그냥 h_input = np.ones()로 해보리면.. 메모리 주소가 새로 할당되면서 Pinned Memory 특성이 풀릴 수 있습니다.
np.copyto(h_input, np.ones(input_shape, dtype=np.float32))

# ONNX 모델은 내부적으로 입력/출력 노드의 이름(name)을 기반으로 데이터를 주고받기 때문에 이름 추출이 필요합니다.
context.set_tensor_address(engine.get_tensor_name(0), int(d_input))
context.set_tensor_address(engine.get_tensor_name(1), int(d_output))

# 최초 Host to Device 전송
# cuda.memcpy가 반복문 밖으로 빠진 이유
# 우리는 모델의 순수 추론 시간을 재고싶습니다. 만약 예열이나 측정 반복문 안에 memcpy를 넣어두면,
# 매번 루프를 돌 때마다 CPU에서 GPU로 데이터를 전송하는 통신 시간이 측정 시간에 포함됩니다.
# 따라서 입력 이미지가 고정(Pinned)되어 있다고 가정하고, 루프 돌기 전에 데이터를 GPU로 딱 한 번 밀어 넣어두고
# 루프 안에서는 오로지 연산(execute_async_v3)만 진행합니다.
cuda.memcpy_htod_async(d_input, h_input, stream)
stream.synchronize()

# 50회 예열
for _ in range(50):
    context.execute_async_v3(stream_handle=stream.handle)
    stream.synchronize()

# 1000회 추론 측정
latencies = []
for _ in range(1000):
    start = time.time()

    context.execute_async_v3(stream_handle=stream.handle)
    stream.synchronize()
    latencies.append((time.time() - start) * 1000)

print('================ [4. Python TensorRT (Zero-Copy) 1000회 벤치마크] ================')
print(f'Avg Latency: {np.mean(latencies):.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print('=================================================================================')
```

<summary><b>5. C++ TensorRT 구현 코드 및 측정 결과</b></summary>

### `YoloTRT.h`
```cpp
#ifndef YOLO_TRT_H
#define YOLO_TRT_H

#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <opencv2/opencv.hpp>
#include <NvInfer.h>
#include <cuda_runtime_api.h>

// TensorFlow나 Pytorch와 달리 TensorRT는 Logger를 기본 내장하지 않고 사용자에게 구현을 강제합니다. 그래서 Logger 선언이 필요합니다.
// override: 부모 클래스인 nvinfer1::ILogger에 있는 가상 함수를 재정의하고 있음을 컴파일러에게 명시하는 키워드입니다.
// noexcept: 이 함수는 절대 예외를 밖으로 던지지 않는다고 컴파일러에게 보장하는 선언입니다.
// C++과 TensorRT 내부 C API가 돌아갈 때 예외로 인한 비정상 종료를 막기 위해서 TensorRT 콜백 함수에는 필수로 붙습니다.
// void log는 내가 호출하는 게 아니라 이벤트가 터졌을 때 TensorRT가 알아서 불러주기 때문에 콜백함수가 맞습니다.
class TRTLogger : public nvinfer1::ILogger {
    void log(Severity severity, const char* msg) noexcept override {
        if (severity <= Severity::kWARNING) {
            std::cout << "[TensorRT] " << msg << std::endl;
        }
    }
};

class YoloTRT {
public:
    YoloTRT(const std::string& enginePath);
    ~YoloTRT();
    void inferOnly();

private:
    TRTLogger gLogger;
    // nvinfer1이 인터페이스라서 인스턴스를 직접 생성할 수 없기 때문에 오직 포인터로만 다뤄야 합니다.
    nvinfer1::IRuntime* runtime = nullptr;
    nvinfer1::ICudaEngine* engine = nullptr;
    nvinfer1::IExecutionContext* context = nullptr;

    // CPU가 GPU에게 작업을 던질 통로(Stream)를 의미합니다.
    cudaStream_t stream = nullptr;

    // Device(GPU)의 입출력 포인터입니다.
    void* d_input = nullptr;
    void* d_output = nullptr;

    size_t inputSize = 1 * 3 * 640 * 640 * sizeof(float);
    size_t outputSize = 1 * 84 * 8400 * sizeof(float);
};
#endif // YOLO_TRT_H
```

### `YoloTRT.cpp`
```cpp
#include "YoloTRT.h"

YoloTRT::YoloTRT(const std::string& enginePath) {
    // ios: Input/Output Stream
    // C++에서 파일 입출력을 담당하는 스트림(Stream) 라이브러리를 의미하고,
    // binary는 이 파일을 줄바꿈 변환을 거치지 않고 순수 이진 데이터를 통째로 읽어라는 뜻입니다.
    std::ifstream file(enginePath, std::ios::binary);
    if (!file.good()) {
        std::cerr << "엔진 파일을 찾을 수 없습니다: " << enginePath << std::endl;
        return; 
    }

    file.seekg(0, file.end); // 파일 커서를 맨 끝(end)으로 이동
    size_t size = file.tellg(); // 현재 커서의 위치(end)는 파일의 전체 바이트 크기이므로, 이것을 변수에 저장합니다.
    file.seekg(0, file.beg); // 파일 커서를 다시 맨 처음(beg)으로 이동

    // 파일 크기만큼의 빈 공간을 가진 1차원 벡터 생성
    // char로 한 이유?
    // C++에서 바이트 크기의 데이터 단위를 표현하기에 char가 가장 적합하기 때문입니다.
    std::vector<char> engineData(size);
    // engineData.data()를 통해 벡터가 메모리에 할당된 시작 주소를 얻어내고,
    // file에 들어있던 바이너리 데이터를 그 주소부터 size만큼 통째로 읽어와서 복사합니다.
    file.read(engineData.data(), size);
    file.close();

    runtime = nvinfer1::createInferRuntime(gLogger);

    // engineData에는 .engine의 바이너리가 1차원 벡터로 담겨있는데, .data()를 써서 1차원 벡터의 주소 시작값을 넘겨주고,
    // 두 번째 인자인 size를 줌으로써 주소 시작값부터 size만큼의 데이터를 역직렬화해서 엔진을 생성합니다.
    engine = runtime->deserializeCudaEngine(engineData.data(), size);
    context = engine->createExecutionContext();

    // 통로(Stream) 생성
    cudaStreamCreate(&stream);

    // inputSize, outputSize만큼의 메모리를 미리 동적 할당합니다.
    // 그리고 그 할당된 메모리의 시작 주소를 d_input, d_output에 담아줍니다.
    cudaMalloc(&d_input, inputSize);
    cudaMalloc(&d_output, outputSize);

    const char* inputName = engine->getIOTensorName(0);
    const char* outputName = engine->getIOTensorName(1);
    // 모델의 입력과 출력 데이터가 실제로 저장될 메모리 주소(d_input, d_output)를 매핑해줍니다.
    context->setTensorAddress(inputName, d_input);
    context->setTensorAddress(outputName, d_output);
}

// 소멸자
YoloTRT::~YoloTRT() {
    // 추론이 끝나면 모든 자원을 반납합니다.
    if (stream) cudaStreamDestroy(stream);
    if (d_input) cudaFree(d_input);
    if (d_output) cudaFree(d_output);
    if (context) delete context;
    if (engine) delete engine;
    if (runtime) delete runtime;
}

void YoloTRT::inferOnly() {
    // 입력 출력 데이터의 메모리 동적 할당은 다 끝났으니 이제 순수 추론만 진행합니다.
    this->context->enqueueV3(this->stream);

    // 지금 stream의 연산이 끝날 때까지 기다립니다. 동기화.
    cudaStreamSynchronize(this->stream);
}
```

### `main.cpp`
```cpp
#include "YoloTRT.h"
#include <chrono>
#include <iostream>
#include <vector>
#include <numeric>
#include <algorithm>

int main() {
    // YoloTRT 클래스 객체 생성. 엔진 파일의 경로를 매개변수로 던져줍니다!
    // 이렇게 하면 생성자에 있는 함수가 호출되며, 엔진 로드 및 입출력 데이터 메모리 동적 할당 등의 준비 작업이 모두 끝납니다.
    YoloTRT detector("yolov8n.engine");

    // 예열 (GPU 클럭 세팅을 위해 50회 수행)
    for (int i = 0; i < 50; ++i) {
        detector.inferOnly();
    }

    const int iterations = 1000; // 반복횟수는 1000회
    std::vector<double> latencies; // 매 추론 시간을 담을 벡터 생성
    latencies.reserve(iterations); // 불필요한 메모리 동적 할당을 막기 위해, 메모리 공간을 미리 예약합니다.

    // 1000회 반복 측정
    for (int i = 0; i < iterations; ++i) {
        auto start = std::chrono::high_resolution_clock::now();
        detector.inferOnly();
        auto end = std::chrono::high_resolution_clock::now();

        double ms = std::chrono::duration<double, std::milli>(end - start).count();
        latencies.push_back(ms);
    }

    double sum = std::accumulate(latencies.begin(), latencies.end(), 0.0);
    double avg = sum / iterations;

    // min과 max를 구하는 방식이 조금 독특합니다.
    // 별(*)을 붙이는 이유
    // std::min_element가 최솟값 숫자 자체를 반환하는 게 아니라, 최솟값이 위치한 주소값을 반환해주기 위함입니다.
    // 그 주소 안에 든 진짜 숫자(알맹이)를 꺼내기 위해 별(*)을 붙입니다. 이를 역참조 연산자라고 합니다.
    double min = *std::min_element(latencies.begin(), latencies.end());
    double max = *std::max_element(latencies.begin(), latencies.end());

    std::cout << "================ [C++ 순수 TRT 1000회 벤치마크] ================" << std::endl;
    std::cout << "Avg Latency: " << cv::format("%.2f", avg) << " ms" << std::endl;
    std::cout << "Min Latency: " << cv::format("%.2f", min) << " ms" << std::endl;
    std::cout << "Max Latency: " << cv::format("%.2f", max) << " ms" << std::endl;
    std::cout << "==================================================================" << std::endl;
}
```
