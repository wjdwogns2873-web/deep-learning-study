# C++ TensorRT: 배치 환경에서의 최적화 시도 및 시행착오 (Try 1, Try 2, Try 3)

항상 한 장의 프레임씩 처리하던 이전의 실험과 달리 이번엔 배치 단위로 프레임 처리를 해보았습니다.

총 세 번의 시도를 했고, 벤치마킹 결과는 아래와 같습니다.

### 1. 아키텍처별 종합 평균 성능 비교표 (요약용)

| 버전 | 핵심 변경 사항 | FP32 평균 FPS (소요 시간) | FP16 평균 FPS (소요 시간) | INT8 평균 FPS (소요 시간) | 평균 배치 크기 |
| :--- | :--- | :---: | :---: | :---: | :---: |
| **Try 1** | 추론 스레드로 단일 전처리 | 60.77 FPS (205.99초) | **70.03 FPS (178.71초)** | **69.90 FPS (179.06초)** | **4.00장** |
| **Try 2** | 카메라 스레드 4개로 전처리 분산 | **64.20 FPS (194.92초)** | 64.03 FPS (195.47초) | 61.53 FPS (203.45초) | 3.95 ~ 3.99장 |
| **Try 3** | 메모리 미리 할당 및 커스텀 삭제자 추가 | 57.57 FPS (217.37초) | 62.13 FPS (201.45초) | 61.80 FPS (202.60초) | 3.96 ~ 3.99장 |

---

## Try 1이 가장 빨랐던 이유

모든 벤치마킹은 코랩 T4 GPU에서 진행되었습니다.

코랩 T4 환경은 CPU 코어 수가 2개 뿐입니다.

Try 1에서는 무거운 CPU 전처리를 추론 스레드 1개만 수행하므로 코어를 서로 빼앗으려는 경쟁이 적었습니다.

반면 Try 2, 3에서는 4개의 카메라 스레드가 동시에 무거운 전처리 연산을 수행하면서 2개의 CPU 코어를 두고 활발한 컨텍스트 스위칭이 발생했습니다.

## Try 2에 비해 Try 3는 성능이 또 하락한 이유?

Try 2에서는 ``batchBuffers.push_back(std::move(pkt.preprocessedData))``로 소유권만 이동시켰습니다.

반면 Try 3에서는 패킷을 Pool에 반납해 재사용해야 하므로 std::move를 쓰지 못하고 

``batchBuffers.push_back(pkt->preprocessedData)``에서 매 배치마다 4.91MB * 4장 용량의 깊은 복사가 발생했습니다.

이 한 줄이 성능 하락의 결정적 원인인것 같습니다.

## 그럼 Try 3는 쓰레기 코드인가?

그렇진 않습니다.

이전 lock-free 보고서에서도 Object Pool을 쓴 적이 있고, 그에 따른 장점은 메모리 단편화를 막을 수 있다는 점입니다.

```cpp
// 탐지된 박스 정보를 표현할 Detection 구조체를 선언합니다.
struct Detection {
    cv::Rect box; // xmin, ymin, width, height를 담을 box
    float confidence; // 확신도를 담을 confidence
    int classId; // 클래스 아이디를 담을 classId
};

// TensorRT에서는 Logger 구현이 필수적입니다.
// TensorRT 엔진 빌드 및 추론 과정에서 발생하는 경고나 에러 메시지를 출력하기 위한 Logger입니다.
class TRTLogger : public nvinfer1::ILogger {
    void log(Severity severity, const char* msg) noexcept override {
        if (severity <= Severity::kWARNING) {
            std::cout << "[TensorRT] " << msg << std::endl;
        }
    }
};

class YoloTRT {
public:
    YoloTRT(const std::string& enginePath);
    ~YoloTRT();

    // 카메라 스레드에서 각자 병렬로 호출할 수 있도록 static으로 선언합니다.
    static void preprocessSingle(const cv::Mat& srcImg, std::vector<float>& outBuffer);

    // 이미 전처리가 완료된 float 벡터 묶음을 받아 복사 후 곧바로 GPU 추론만 수행합니다.
    std::vector<std::vector<Detection>> detectBatchPreprocessed(
        const std::vector<std::vector<float>>& batchBuffers, 
        const std::vector<cv::Size>& origSizes, 
        float confThreshold = 0.25f, 
        float nmsThreshold = 0.45f
    );

    // 실제로 추론할 detect 함수를 선언합니다.
    // 이미지, conf 임계값과 nms 임계값을 매개변수로 받아서 std::vector<Detection>을 반환합니다.
    std::vector<Detection> detect(const cv::Mat& srcImg, float confThresh = 0.25f, float nmsThresh = 0.45f);

    // 배치 단위로 추론 결과가 나오기 때문에, std::vector<Detection>을 한 번더 벡터로 감싸서
    // std::vector<std::vector<Detection>>이 됩니다.
    // cv::Mat도 벡터로 들어갑니다.
    std::vector<std::vector<Detection>> detectBatch(
        const std::vector<cv::Mat>& srcImgs, 
        float confThreshold = 0.25f, 
        float nmsThreshold = 0.45f
    );

    void inferOnly();

    // 1장당 요소 개수 상수를 외부(카메라 스레드)에서도 버퍼 크기 예약에 쓸 수 있도록 public으로 이동합니다.
    static constexpr size_t kSingleInputElements = 3 * 640 * 640;
private:
    static constexpr int kMaxBatchSize = 4;

    TRTLogger gLogger;
    // nvinfer1이 인터페이스라서 인스턴스를 직접 생성할 수 없기 때문에 오직 포인터로만 다뤄야 합니다.
    nvinfer1::IRuntime* runtime = nullptr;
    nvinfer1::ICudaEngine* engine = nullptr;
    nvinfer1::IExecutionContext* context = nullptr;
    
    // 전처리 함수입니다. 640 640 리사이즈, HWC to CHW, BGR to RGB 등의 전처리 작업을 수행합니다.
    // 배치 인덱스를 받아 h_input의 알맞은 위치에 기록하도록 수정합니다.
    void preprocess(const cv::Mat& srcImg, int batchIdx = 0);

    // 후처리 함수입니다. 원본이미지로 복원하는 과정, nms 작업 등이 있습니다.
    // 여기서도 배치 인덱스를 받아서 h_output의 알맞은 위치에 기록하도록 수정합니다.
    std::vector<Detection> postprocess(int origW, int origH, float confThresh, float nmsThresh, int batchIdx = 0);

    cudaStream_t stream = nullptr;

    // Device(GPU)의 입출력 포인터(주소값)입니다.
    void* d_input = nullptr;
    void* d_output = nullptr;

    // Host(CPU)의 입출력 포인터(주소값)입니다.
    float* h_input = nullptr;
    float* h_output = nullptr;

    // 1장당 요소 개수 상수 (포인터 오프셋 계산용)
    static constexpr size_t kSingleOutputElements = 84 * 8400;

    // 최대 배치 크기 기준으로 메모리 할당 크기 확장
    size_t maxInputSize = kMaxBatchSize * kSingleInputElements * sizeof(float);
    size_t maxOutputSize = kMaxBatchSize * kSingleOutputElements * sizeof(float);

    // 입출력 텐서의 메모리 크기입니다.
    // size_t inputSize = 1 * 3 * 640 * 640 * sizeof(float);
    // size_t outputSize = 1 * 84 * 8400 * sizeof(float);

    // 동적으로 메모리 할당 및 해제를 방지하기 위해 최초 한 번만 생성합니다.
    std::vector<cv::Rect> boxes; // 탐지된 네모박스 정보(cv::Rect)를 담기 위해 std::vector<cv::Rect>를 선언합니다.
    std::vector<float> confidences; // 네모박스들의 확신도를 담기 위해 std::vector<float>을 선언합니다.
    std::vector<int> classIds; // 네모박스들의 클래스 아이디를 담기 위해 std::vector<int>를 선언합니다.
    std::vector<int> indices; // conf, nms 임계값을 넘은 네모박스들의 임계값을 담을 std::vector<int>를 선언합니다.
};

#endif // YOLO_TRT_H
```

### `YoloTRT_orig.cpp`

```cpp
#include "YoloTRT_orig.h"
namespace fs = std::filesystem;

YoloTRT::YoloTRT(const std::string& enginePath) {
    std::ifstream file(enginePath, std::ios::binary);
    if (!file.good()) {
        std::cerr << "엔진 파일을 찾을 수 없습니다: " << enginePath << std::endl;
        return; 
    }

    // 1. 맨 앞 4바이트를 읽어 Ultralytics 메타데이터 크기 확인
    uint32_t metaLen = 0;
    file.read(reinterpret_cast<char*>(&metaLen), sizeof(uint32_t));

    // 2. 전체 파일 크기 계산
    file.seekg(0, file.end);
    size_t totalSize = file.tellg();

    // 3. 메타데이터 길이만큼 건너뛴 위치(4 + metaLen)로 커서 이동
    size_t engineOffset = sizeof(uint32_t) + metaLen;
    size_t engineSize = totalSize - engineOffset;
    file.seekg(engineOffset, file.beg);

    // 4. 진짜 TensorRT 바이너리 영역만 읽어오기
    std::vector<char> engineData(engineSize);
    file.read(engineData.data(), engineSize);
    file.close();

    runtime = nvinfer1::createInferRuntime(gLogger);
    engine = runtime->deserializeCudaEngine(engineData.data(), engineSize);
    context = engine->createExecutionContext();

    cudaStreamCreate(&stream);

    cudaMalloc(&d_input, maxInputSize);
    cudaMalloc(&d_output, maxOutputSize);

    cudaHostAlloc((void**)&h_input, maxInputSize, cudaHostAllocDefault);
    cudaHostAlloc((void**)&h_output, maxOutputSize, cudaHostAllocDefault);

    boxes.reserve(8400);
    confidences.reserve(8400);
    classIds.reserve(8400);
    indices.reserve(8400);

    const char* inputName = engine->getIOTensorName(0);
    const char* outputName = engine->getIOTensorName(1);
    context->setTensorAddress(inputName, d_input);
    context->setTensorAddress(outputName, d_output);
}

YoloTRT::~YoloTRT() {
    // 추론이 끝나면, 메모리 누수를 막기 위해 모든 자원을 반납하여 할당 해제합니다.
    if (stream) cudaStreamDestroy(stream);
    if (d_input) cudaFree(d_input);
    if (d_output) cudaFree(d_output);

    if (h_input) cudaFreeHost(h_input);
    if (h_output) cudaFreeHost(h_output);

    if (context) delete context;
    if (engine) delete engine;
    if (runtime) delete runtime;
}

void YoloTRT::preprocess(const cv::Mat& srcImg, int batchIdx) {
    cv::Mat resizedImg;
    cv::resize(srcImg, resizedImg, cv::Size(640, 640));
    cv::cvtColor(resizedImg, resizedImg, cv::COLOR_BGR2RGB);

    cv::Mat floatImg;
    resizedImg.convertTo(floatImg, CV_32FC3, 1.0 / 255.0);

    int channelSize = 640 * 640;
    float* data = reinterpret_cast<float*>(floatImg.data);

    // batchIdx만큼 포인터 주소를 뒤로 밀어줍니다. 0번 이미지는 0에서 시작, 1번 이미지는 1,228,800(3 * 640 * 640)에서 시작..
    float* targetInput = h_input + (batchIdx * kSingleInputElements);

    for (int i = 0; i < channelSize; ++i) {
        targetInput[i]                   = data[i * 3 + 0]; // R
        targetInput[i + channelSize]     = data[i * 3 + 1]; // G
        targetInput[i + 2 * channelSize] = data[i * 3 + 2]; // B
    }
}

std::vector<Detection> YoloTRT::postprocess(int origW, int origH, float confThresh, float nmsThresh, int batchIdx) {
    boxes.clear();
    confidences.clear();
    classIds.clear();
    indices.clear();

    float scaleX = static_cast<float>(origW) / 640.0f;
    float scaleY = static_cast<float>(origH) / 640.0f;
    // batchIdx만큼 h_output의 포인터 주소를 뒤로 밀어줍니다. 0번 결과는 0에서 시작, 1번 결과는 705,600(84 * 8400)에서 시작..
    const float* targetOutput = h_output + (batchIdx * kSingleOutputElements);

    for (int i = 0; i < 8400; ++i) {
        float cx = targetOutput[0 * 8400 + i];
        float cy = targetOutput[1 * 8400 + i];
        float w  = targetOutput[2 * 8400 + i];
        float h  = targetOutput[3 * 8400 + i];

        float maxScore = 0.0f;
        int maxClassId = -1;
        for (int c = 0; c < 80; ++c) {
            float score = targetOutput[(c + 4) * 8400 + i];
            if (score > maxScore) {
                maxScore = score;
                maxClassId = c;
            }
        }

        if (maxScore > confThresh) {
            float x1 = (cx - w / 2.0f) * scaleX;
            float y1 = (cy - h / 2.0f) * scaleY;
            float boxW = w * scaleX;
            float boxH = h * scaleY;

            cv::Rect box(x1, y1, boxW, boxH);
            cv::Rect bounds(0, 0, origW, origH);
            box &= bounds;

            boxes.emplace_back(box);
            confidences.emplace_back(maxScore);
            classIds.emplace_back(maxClassId);
        }
    }

    cv::dnn::NMSBoxes(boxes, confidences, confThresh, nmsThresh, indices);

    std::vector<Detection> detections;
    detections.reserve(indices.size());
    for (int idx : indices) {
        detections.push_back({ boxes[idx], confidences[idx], classIds[idx] });
    }

    return detections;
}

std::vector<Detection> YoloTRT::detect(const cv::Mat& srcImg, float confThresh, float nmsThresh) {
    if (srcImg.empty()) return {};

    // 동적 배치 엔진에서도 안전하게 동작하도록 입력 상태를 1장으로 명시합니다.
    const char* inputName = engine->getIOTensorName(0);
    context->setInputShape(inputName, nvinfer1::Dims4{1, 3, 640, 640});

    preprocess(srcImg, 0);

    size_t singleInputBytes = kSingleInputElements * sizeof(float);
    size_t singleOutputBytes = kSingleOutputElements * sizeof(float);

    cudaMemcpyAsync(d_input, h_input, singleInputBytes, cudaMemcpyHostToDevice, stream);
    context->enqueueV3(stream);
    cudaMemcpyAsync(h_output, d_output, singleOutputBytes, cudaMemcpyDeviceToHost, stream);
    cudaStreamSynchronize(stream);

    return postprocess(srcImg.cols, srcImg.rows, confThresh, nmsThresh, 0);
}

std::vector<std::vector<Detection>> YoloTRT::detectBatch(const std::vector<cv::Mat>& srcImgs, float confThresh, float nsmThresh) {
    int batchSize = static_cast<int>(srcImgs.size());
    if (batchSize == 0 || batchSize > kMaxBatchSize) return {};

    // 배치 사이즈만큼 입력 shape을 설정합니다. 꼭 4(maxBatchSize)가 아닐 수도 있습니다.
    const char* inputName = engine->getIOTensorName(0);
    context->setInputShape(inputName, nvinfer1::Dims4{batchSize, 3, 640, 640});

    // 배치에 들어온 이미지를 h_input 버퍼에 0번 부터 차례대로 전처리해서 채워넣습니다.
    for (int b = 0; b < batchSize; ++b) {
        preprocess(srcImgs[b], b);
    }

    // 실제 들어온 배치사이즈 만큼만 바이트 크기를 계산해서 복사합니다. cudaMemcpyAsync
    size_t currentInputBytes = batchSize * kSingleInputElements * sizeof(float);
    size_t currentOutputBytes = batchSize * kSingleOutputElements * sizeof(float);

    cudaMemcpyAsync(d_input, h_input, currentInputBytes, cudaMemcpyHostToDevice, stream);
    context->enqueueV3(stream);
    cudaMemcpyAsync(h_output, d_output, currentOutputBytes, cudaMemcpyDeviceToHost, stream);
    cudaStreamSynchronize(stream);

    // 이 함수의 반환형입니다. 하나의 프레임(이미지)은 std::vector<Detection>을 배출하는데, 배치 안에 여러장의 프레임이 있으므로
    // std::vector<Detection>을 std::vector로 감싸야 합니다.
    std::vector<std::vector<Detection>> batchResults;
    batchResults.reserve(batchSize); // 배치 사이즈만큼 메모리 공간을 예약합니다.

    // 배치의 각 이미지마다 후처리를 한 결과를 벡터(batchResults)에 담습니다.
    for (int b = 0; b < batchSize; ++b) {
        batchResults.push_back(postprocess(srcImgs[b].cols, srcImgs[b].rows, confThresh, nsmThresh, b));
    }

    return batchResults;
}

void YoloTRT::preprocessSingle(const cv::Mat& srcImg, std::vector<float>& outBuffer) {
    // 출력 버퍼가 정확한 크기(3*640*640)를 갖도록 합니다.
    if (outBuffer.size() != kSingleInputElements) {
        outBuffer.resize(kSingleInputElements);
    }

    cv::Mat resizedImg;
    cv::resize(srcImg, resizedImg, cv::Size(640, 640));
    cv::cvtColor(resizedImg, resizedImg, cv::COLOR_BGR2RGB);

    cv::Mat floatImg;
    resizedImg.convertTo(floatImg, CV_32FC3, 1.0 / 255.0);

    int channelSize = 640 * 640;
    const float* data = reinterpret_cast<const float*>(floatImg.data);
    float* target = outBuffer.data();

    for (int i = 0; i < channelSize; ++i) {
        target[i]                   = data[i * 3 + 0]; // R
        target[i + channelSize]     = data[i * 3 + 1]; // G
        target[i + channelSize * 2] = data[i * 3 + 2]; // B
    }
}

std::vector<std::vector<Detection>> YoloTRT::detectBatchPreprocessed(
    const std::vector<std::vector<float>>& batchBuffers, 
    const std::vector<cv::Size>& origSizes, 
    float confThresh, 
    float nmsThresh
) {
    int batchSize = static_cast<int>(batchBuffers.size());
    if (batchSize == 0 || batchSize > kMaxBatchSize) return {};

    const char* inputName = engine->getIOTensorName(0);
    context->setInputShape(inputName, nvinfer1::Dims4{batchSize, 3, 640, 640});

    // CPU 전처리 연산(resize, cvtColor 등)이 완전히 사라지고, 
    // 이미 완성된 float 배열을 Pinned memory(h_input)의 알맞은 위치에 고속 메모리 복사(memcpy)만 수행합니다.
    size_t singleBytes = kSingleInputElements * sizeof(float);
    for (int b = 0; b < batchSize; ++b) {
        float* destPtr = h_input + (b * kSingleInputElements);
        std::memcpy(destPtr, batchBuffers[b].data(), singleBytes);
    }

    size_t currentInputBytes = batchSize * singleBytes;
    size_t currentOutputBytes = batchSize * kSingleOutputElements * sizeof(float);

    // GPU 메모리 전송 및 추론 실행
    cudaMemcpyAsync(d_input, h_input, currentInputBytes, cudaMemcpyHostToDevice, stream);
    context->enqueueV3(stream);
    cudaMemcpyAsync(h_output, d_output, currentOutputBytes, cudaMemcpyDeviceToHost, stream);
    cudaStreamSynchronize(stream);

    // 후처리 수행
    std::vector<std::vector<Detection>> batchResults;
    batchResults.reserve(batchSize);

    for (int b = 0; b < batchSize; ++b) {
        batchResults.push_back( postprocess(origSizes[b].width, origSizes[b].height, confThresh, nmsThresh, b) );
    }

    return batchResults;
}

void YoloTRT::inferOnly() {
    context->enqueueV3(stream);
    cudaStreamSynchronize(stream);
}
```

Try 1:

카메라 4대와 추론하는 엔진 1개가 있습니다.

카메라 4대는 각자 프레임을 수집해서 버퍼(buffer_)에 넣고, 이 버퍼에 4개가 쌓이거나 15ms가 지나면 배치단위로 묶어서 추론 스레드에게 추론을 맡깁니다.

추론 스레드는 순수 추론만 하는 게 아니라 전처리, 순수 추론, 후처리 작업을 모두 도맡아서 합니다.

### `batchCollector_orig.cpp`

```cpp
#include <iostream>

#include <thread>

#include <condition_variable>
#include <chrono>
#include <memory>
#include <atomic>
#include <opencv2/opencv.hpp>

// 실제 이미지(cv::Mat)의 포인터가 담긴 구조체입니다.
struct FramePacket {
    int streamId; // 카메라 ID
    int frameNumber; // 프레임 번호(ex: 1 ~ 3129)
    std::shared_ptr<cv::Mat> imagePtr; // 제로카피를 위한 이미지의 포인터
};

// 배치를 모으고(push), 추론하는(popBatch) 클래스입니다.
class DynamicBatchCollector {
private:
    std::vector<FramePacket> buffer_; // 프레임 패킷을 배치 단위로 모을 벡터입니다.
    size_t maxBatchSize_; // 배치의 최대 개수입니다. 여기서는 4로 했습니다.
    std::chrono::milliseconds timeout_; // 한 배치에 4개가 쌓이지 않아도 일정 시간이 지나면 태워서 추론에 보내도록 타임아웃 시간을 설정합니다.
    std::mutex mtx_; // 여러 스레드의 동시 접근을 막는 mutex입니다.

    // 버퍼(buffer_)에 데이터가 들어올 때까지 추론 스레드를 재우고 깨우는 조건변수입니다.
    std::condition_variable cvNotEmpty_;
    // 버퍼에 빈 공간이 생길 때까지 카메라 스레드를 재우는 조건변수입니다.
    std::condition_variable cvNotFull_;
public:
    // 생성자: 멤버 초기화 리스트(:)를 사용해 변수들을 초기화합니다.
    // 불필요한 기본 생성 후 대입 과정을 생략해서 성능을 높입니다.
    DynamicBatchCollector(size_t maxBatchSize, int timeoutMs)
        : maxBatchSize_(maxBatchSize), timeout_(std::chrono::milliseconds(timeoutMs)) {
        buffer_.reserve(maxBatchSize_); // 동적 할당 방지를 위해 메모리 공간을 미리 예약합니다.
    }

    // 프레임 패킷을 우측값 참조(&&)로 받습니다. 소유권 이동(std::move)를 하기 위해서입니다.
    bool push(FramePacket&& packet, const std::atomic<bool>& stopSignal) {
        std::unique_lock<std::mutex> lock(mtx_); // 다른 스레드가 접근하지 못하게 자물쇠를 걸어 잠급니다.

        // 버퍼(buffer_)가 가득차지 않았다면(공간이 있다면) 밑으로 통과하게 합니다.
        // 공간이 없다면, 공간이 생길 때까지 스레드를 재워서 CPU 낭비를 막습니다.
        cvNotFull_.wait(lock, [this, &stopSignal]() {
            return buffer_.size() < maxBatchSize_ || stopSignal.load();
        });

        if (stopSignal.load()) return false;

        buffer_.push_back(std::move(packet)); // 소유권 이동함으로써, 복사비용 없이 버퍼(buffer_)에 적재합니다.
        cvNotEmpty_.notify_one(); // 버퍼에 프레임 패킷이 들어왔으니 추론 스레드를 깨웁니다.
        return true;
    }

    // 바꿔치기(swap)하기 위해 빈 벡터를 가져옵니다.
    bool popBatch(std::vector<FramePacket>& outBatch, const std::atomic<bool>& stopSignal) {
        std::unique_lock<std::mutex> lock(mtx_); // 다른 스레드가 접근하지 못하게 자물쇠를 걸어 잠급니다.

        // 버퍼(buffer_)가 비어있지 않다면(프레임이 존재한다면) 밑으로 통과하게 합니다.
        // 프레임이 존재하지 않는다면 생길 때까지 스레드를 재워서 CPU 낭비를 막습니다.
        cvNotEmpty_.wait(lock, [this, &stopSignal]() {
            return !buffer_.empty() || stopSignal.load();
        });

        // 버퍼(buffer_)가 비어있고, 종료신호가 켜졌다면 false를 반환합니다.
        if (buffer_.empty() && stopSignal.load()) return false;

        // 버퍼(buffer_) 안에 4개 이상(4개를 초과할 수 없기 때문에 사실상 4개가 맥시멈)이거나 지정한 시간(timeout_)이 지났을 때 밑으로 통과하게 합니다.
        cvNotEmpty_.wait_for(lock, timeout_, [this, &stopSignal]() {
            return buffer_.size() >= maxBatchSize_ || stopSignal.load();
        });

        // 버퍼(buffer_)에 모인 프레임들의 소유권을 통째로 outBatch로 이동(swap)시킵니다.
        // outBatch는 빈 벡터였으므로, 바꿔치기 함으로써 buffer_는 빈 벡터가 됩니다.
        outBatch.swap(buffer_);

        // 버퍼를 완전히 비웠으니 대기 중이던 카메라 스레드들을 모두 깨웁니다.
        cvNotFull_.notify_all();
        return true;
    }

    void notifyStop() {
        std::lock_guard<std::mutex> lock(mtx_);
        cvNotEmpty_.notify_all();
        cvNotFull_.notify_all();
    }
};

// 카메라 4대가 쓰는 스레드입니다. VideoCapture 객체에서 나온 프레임을 프레임 패킷 구조체에 담아서 push합니다.
void cameraStreamThread(
    int streamId, 
    const std::string& videoPath, 
    DynamicBatchCollector& collector, 
    const std::atomic<bool>& stopSignal
) {
    cv::VideoCapture cap(videoPath);
    if (!cap.isOpened()) {
        std::cerr << "[카메라 " << streamId << "] 비디오 열기 실패\n";
        return;
    }

    int frameIdx = 0;
    while (!stopSignal.load()) {
        auto framePtr = std::make_shared<cv::Mat>(); // 스마트 포인터 그릇을 만듭니다.
        cap >> *framePtr; // 역참조하여서 VideoCapture에서 나온 프레임을 받습니다. 알맹이에 접근합니다.

        if (framePtr->empty()) break;

        frameIdx++;
        // shared_ptr의 참조 카운트 증가 비용조차 아끼기 위해 소유권을 이동합니다.
        FramePacket packet{ streamId, frameIdx, std::move(framePtr) };

        // push 함수에서 packet을 우측값 참조(&&)로 받고있기 때문에, std::move()로 넘겨줍니다.
        if (!collector.push(std::move(packet), stopSignal)) break;
    }
    cap.release();
}

// 배치 단위로 들어온 프레임을 consumer가 추론하는 스레드입니다.
void batchInferenceThread(
    YoloTRT& detector, 
    DynamicBatchCollector& collector, 
    const std::atomic<bool>& stopSignal, 
    int& totalProcessedFrames, 
    int& totalBatches, 
    const size_t maxBatchSize
) {
    std::vector<FramePacket> batch; // 바꿔치기(swap)하기 위해 벡터 껍데기를 준비해둡니다.
    std::vector<cv::Mat> batchImages; // 추론시 cv::Mat만 배치로 묶어서 보내기 위해 벡터를 준비합니다.
    batch.reserve(maxBatchSize); // 동적 할당을 막기 위해 미리 공간을 예약합니다.
    batchImages.reserve(maxBatchSize);

    while (true) {
        // 새로운 배치를 받기 전에 항상 비워줍니다.
        batch.clear();
        batchImages.clear();

        if (!collector.popBatch(batch, stopSignal)) break; // 바꿔치기(swap) 되었습니다.

        // cv::Mat의 스마트 포인터를 역참조(*)하여 알맹이만 batchImages 벡터에 담습니다.
        // cv::Mat은 대입연산자를 쓰거나 push_back을 할 때 데이터 통째로 복사하는 깊은복사가 아닌, 
        // 주소만 공유하는 얕은 복사가 됩니다. 그렇기 때문에 std::move를 쓰지 않았더라도 복사비용이 크지 않습니다.
        for (const auto& pkt : batch) {
            batchImages.push_back(*pkt.imagePtr);
        }

        // 담은 알맹이는 배치단위로 추론하는 데에 쓰입니다.
        // 실제 여기서의 자료형은 std::vector<std::vector<Detection>>이지만, 길기 때문에 auto로 했습니다.
        auto batchDetections = detector.detectBatch(batchImages, 0.25f, 0.45f);

        // 배치 개수 만큼 바운딩박스 그리기 작업을 해줍니다.
        // 여기 역시 역참조(*)하여 알맹이를 가져옵니다. cv::rectangle 함수에서는 알맹이를 필요로 합니다.
        for (size_t i = 0; i < batch.size(); ++i) {
            for (const auto& det : batchDetections[i]) {
                cv::rectangle(*batch[i].imagePtr, det.box, cv::Scalar(0, 255, 0), 2);
            }
        }

        totalProcessedFrames += static_cast<int>(batch.size());
        totalBatches++;
    }
}

int main() {
    std::string engineName = "yolov8n.dynamic_b4.int8.engine";
    YoloTRT detector(engineName);
    std::cout << "사용한 엔진: " << engineName << std::endl;

    // 테스트용으로 같은 비디오를 4개 사용합니다. 카메라도 4대라고 가정하는 겁니다.
    std::vector<std::string> videoPaths = {
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4"
    };

    const size_t maxBatchSize = 4;
    const int timeoutMs = 15; // 15ms 동안 배치에 4개가 채워지지 않으면 지체하지 않고 태워서 추론으로 보냅니다.

    // Warm up 합니다.
    std::vector<cv::Mat> dummyBatch(maxBatchSize, cv::Mat::zeros(640, 640, CV_8UC3));
    for (int i = 0; i < 30; ++i) {
        detector.detectBatch(dummyBatch, 0.25f, 0.45f);
    }

    DynamicBatchCollector collector(maxBatchSize, timeoutMs);
    std::atomic<bool> stopSignal{false};
    int totalProcessedFrames = 0;
    int totalBatches = 0;

    auto totalStart = std::chrono::high_resolution_clock::now();

    // 추론 스레드 시작합니다. consumer
    std::thread inferencer(
        batchInferenceThread, 
        std::ref(detector), 
        std::ref(collector), 
        std::ref(stopSignal), 
        std::ref(totalProcessedFrames), 
        std::ref(totalBatches), 
        maxBatchSize
    );

    // 총 4개의 카메라 스레드를 시작합니다. producer
    std::vector<std::thread> cameraThreads;
    for (int i = 0; i < static_cast<int>(videoPaths.size()); ++i) {
        cameraThreads.emplace_back(cameraStreamThread, i, videoPaths[i], std::ref(collector), std::ref(stopSignal));
    }

    // 모든 카메라 스트림이 끝날 때까지 대기합니다.
    for (auto& t : cameraThreads) {
        if (t.joinable()) t.join();
    }

    // 모든 카메라 종료 후 추론 스레드를 깨워서 종료시킵니다.
    stopSignal = true;
    collector.notifyStop();
    if (inferencer.joinable()) inferencer.join();

    auto totalEnd = std::chrono::high_resolution_clock::now();
    double totalSec = std::chrono::duration<double>(totalEnd - totalStart).count();

    std::cout << "총 처리 프레임 수: " << totalProcessedFrames << " 프레임" << std::endl;
    std::cout << "총 배치 횟수: " << totalBatches << "회 (평균 배치 크기: "
              << cv::format("%.2f", static_cast<double>(totalProcessedFrames) / totalBatches) << "장)" << std::endl;
    std::cout << "총 소요 시간: " << cv::format("%.2f", totalSec) << "초" << std::endl;
    std::cout << "평균 FPS: " << cv::format("%.1f", totalProcessedFrames / totalSec) << " FPS" << std::endl;
}
```

Try 2:

Try 2에서는 Try 1과의 차별점이 있습니다.

Try 1에서는 전처리 추론 후처리를 모두 추론 스레드가 했다면, Try 2에서는 4개의 카메라 스레드가 받은 프레임을 각자 전처리를 하고, 

전처리가 완료된 float 버퍼를 추론 스레드에게 넘겨주는 방식입니다.

즉, 추론 스레드는 순수 추론과 후처리만 하면 되는 상황입니다.

이렇게 하면 성능이 Try 1보다 향상될 거라고 예상했지만, 결과는 그렇지 않았습니다.

FP32만 성능이 미세하게 향상됐고, 나머지 FP16, INT8은 구버전보다 성능이 떨어졌습니다. 그 이유는 아래와 같습니다.

코랩 T4 무료 환경의 하드웨어 제약인데, GPU는 T4를 주지만, CPU는 2코어만 할당됩니다.

현재 코드에서는 카메라 스레드 4개 + 추론 스레드 1개 + 메인 스레드 1개. 총 6개의 스레드가 동시에 돌고 있습니다.

CPU 코어는 2개 뿐인데, 4개의 카메라 스레드가 동시에 무거운 cv::resize, cvtColor 등의 전처리 연산을 수행하면서 
2개의 코어를 두고 뺏고 뺏기는 컨텍스트 스위칭 오버헤드가 발생했습니다.

```cpp
#include <iostream>
#include <vector>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <chrono>
#include <memory>

#include <opencv2/opencv.hpp>

// 흐름 정리
/**
프로그램이 실행되면 4개의 카메라 스레드와 1개의 추론 스레드가 가동됩니다.
카메라 스레드 4개는 각각 비디오에서 프레임을 받아서 전처리(preprocessSingle)를 수행하고, 
전처리된 CHW 텐서 데이터(std::vector<float> preprocessdData)를 담아둡니다. 

추론 스레드는 배치에 4개가 가득 차거나, 15ms가 지나면 배치에 있는 프레임을 가지고 추론 및 후처리를 진행합니다.
 */

 /*

 */

// 실제 이미지(cv::Mat)의 포인터가 담긴 구조체입니다.
struct FramePacket {
    int streamId; // 카메라 ID
    int frameNumber; // 프레임 번호(ex: 1 ~ 3129)
    std::shared_ptr<cv::Mat> imagePtr; // 제로카피를 위한 이미지의 포인터
    std::vector<float> preprocessedData; // 카메라 스레드가 미리 전처리해둔 CHW 텐서 데이터
    cv::Size origSize; // 후처리 좌표 복원용 원본 해상도
};

// 배치를 모으고(push), 추론하는(popBatch) 클래스입니다.
class DynamicBatchCollector {
private:
    std::vector<FramePacket> buffer_; // 프레임 패킷을 배치 단위로 모을 벡터입니다.
    size_t maxBatchSize_; // 배치의 최대 개수입니다. 여기서는 4로 했습니다.
    std::chrono::milliseconds timeout_; // 한 배치에 4개가 쌓이지 않아도 일정 시간이 지나면 태워서 추론에 보내도록 타임아웃 시간을 설정합니다.
    std::mutex mtx_; // 여러 스레드의 동시 접근을 막는 mutex입니다.

    // 버퍼(buffer_)에 데이터가 들어올 때까지 추론 스레드를 재우고 깨우는 조건변수입니다.
    std::condition_variable cvNotEmpty_;
    // 버퍼에 빈 공간이 생길 때까지 카메라 스레드를 재우는 조건변수입니다.
    std::condition_variable cvNotFull_;
public:
    // 생성자: 멤버 초기화 리스트(:)를 사용해 변수들을 초기화합니다.
    // 불필요한 기본 생성 후 대입 과정을 생략해서 성능을 높입니다.
    DynamicBatchCollector(size_t maxBatchSize, int timeoutMs)
        : maxBatchSize_(maxBatchSize), timeout_(std::chrono::milliseconds(timeoutMs)) {
        buffer_.reserve(maxBatchSize_); // 동적 할당 방지를 위해 메모리 공간을 미리 예약합니다.
    }

    // 프레임 패킷을 우측값 참조(&&)로 받습니다. 소유권 이동(std::move)를 하기 위해서입니다.
    bool push(FramePacket&& packet, const std::atomic<bool>& stopSignal) {
        std::unique_lock<std::mutex> lock(mtx_); // 다른 스레드가 접근하지 못하게 자물쇠를 걸어 잠급니다.

        // 버퍼(buffer_)가 가득차지 않았다면(공간이 있다면) 밑으로 통과하게 합니다.
        // 공간이 없다면, 공간이 생길 때까지 스레드를 재워서 CPU 낭비를 막습니다.
        cvNotFull_.wait(lock, [this, &stopSignal]() {
            return buffer_.size() < maxBatchSize_ || stopSignal.load();
        });

        if (stopSignal.load()) return false;

        buffer_.push_back(std::move(packet)); // 소유권 이동함으로써, 복사비용 없이 버퍼(buffer_)에 적재합니다.
        cvNotEmpty_.notify_one(); // 버퍼에 프레임 패킷이 들어왔으니 추론 스레드를 깨웁니다.
        return true;
    }

    // 바꿔치기(swap)하기 위해 빈 벡터를 가져옵니다.
    bool popBatch(std::vector<FramePacket>& outBatch, const std::atomic<bool>& stopSignal) {
        std::unique_lock<std::mutex> lock(mtx_); // 다른 스레드가 접근하지 못하게 자물쇠를 걸어 잠급니다.

        // 버퍼(buffer_)가 비어있지 않다면(프레임이 존재한다면) 밑으로 통과하게 합니다.
        // 프레임이 존재하지 않는다면 생길 때까지 스레드를 재워서 CPU 낭비를 막습니다.
        cvNotEmpty_.wait(lock, [this, &stopSignal]() {
            return !buffer_.empty() || stopSignal.load();
        });

        // 버퍼(buffer_)가 비어있고, 종료신호가 켜졌다면 false를 반환합니다.
        if (buffer_.empty() && stopSignal.load()) return false;

        // 버퍼(buffer_) 안에 4개 이상(4개를 초과할 수 없기 때문에 사실상 4개가 맥시멈)이거나 지정한 시간(timeout_)이 지났을 때 밑으로 통과하게 합니다.
        cvNotEmpty_.wait_for(lock, timeout_, [this, &stopSignal]() {
            return buffer_.size() >= maxBatchSize_ || stopSignal.load();
        });

        // 버퍼(buffer_)에 모인 프레임들의 소유권을 통째로 outBatch로 이동(swap)시킵니다.
        outBatch.swap(buffer_);

        // 버퍼를 완전히 비웠으니 대기 중이던 카메라 스레드들을 모두 깨웁니다.
        cvNotFull_.notify_all();
        return true;
    }

    void notifyStop() {
        std::lock_guard<std::mutex> lock(mtx_);
        cvNotEmpty_.notify_all();
        cvNotFull_.notify_all();
    }
};

// 카메라 4대가 쓰는 스레드입니다. VideoCapture 객체에서 나온 프레임을 프레임 패킷 구조체에 담아서 push합니다.
void cameraStreamThread(
    int streamId, 
    const std::string& videoPath, 
    DynamicBatchCollector& collector, 
    const std::atomic<bool>& stopSignal
) {
    cv::VideoCapture cap(videoPath);
    if (!cap.isOpened()) {
        std::cerr << "[카메라 " << streamId << "] 비디오 열기 실패\n";
        return;
    }

    int frameIdx = 0;
    while (!stopSignal.load()) {
        auto framePtr = std::make_shared<cv::Mat>(); // 스마트 포인터 그릇을 만듭니다.
        cap >> *framePtr; // 역참조하여서 VideoCapture에서 나온 프레임을 받습니다. 알맹이에 접근합니다.

        if (framePtr->empty()) break;

        frameIdx++;

        // 추론 스레드가 혼자 하던 무거운 CPU 전처리 연산을 4개의 카메라 스레드가 각자 병렬로 수행합니다.
        std::vector<float> prepBuffer(YoloTRT::kSingleInputElements);
        YoloTRT::preprocessSingle(*framePtr, prepBuffer);

        // std::move(framePtr)을 하게 되면 framePtr은 빈 껍데기가 됩니다.
        // 빈 껍데기가 되기 전에 size를 저장해 둡니다.
        cv::Size origSize = framePtr->size();

        // shared_ptr의 참조 카운트 증가 비용조차 아끼기 위해 소유권을 이동합니다.
        FramePacket packet{
            streamId, 
            frameIdx, 
            std::move(framePtr), 
            std::move(prepBuffer), // 전처리 버퍼의 소유권을 복사 없이 이동
            // cv::Size(framePtr->cols, framePtr->rows)    
            origSize
        };

        // push 함수에서 packet을 우측값 참조(&&)로 받고있기 때문에, std::move()로 넘겨줍니다.
        if (!collector.push(std::move(packet), stopSignal)) break;
    }
    cap.release();
}

// 배치 단위로 들어온 프레임을 consumer가 추론하는 스레드입니다.
void batchInferenceThread(
    YoloTRT& detector, 
    DynamicBatchCollector& collector, 
    const std::atomic<bool>& stopSignal, 
    int& totalProcessedFrames, 
    int& totalBatches, 
    const size_t maxBatchSize
) {
    std::vector<FramePacket> batch;
    std::vector<std::vector<float>> batchBuffers;
    // 후처리할 때 원본 크기로 복원이 필요하므로, cv::Size를 담을 벡터를 하나 선언합니다.
    std::vector<cv::Size> origSizes;

    batch.reserve(maxBatchSize);
    batchBuffers.reserve(maxBatchSize);
    origSizes.reserve(maxBatchSize);

    while (true) {
        batch.clear();
        batchBuffers.clear();
        origSizes.clear();

        if (!collector.popBatch(batch, stopSignal)) break;

        for (auto& pkt : batch) {
            batchBuffers.push_back(std::move(pkt.preprocessedData));
            origSizes.push_back(pkt.origSize);
        }

        // 전처리 과정이 생략된 배치 추론 함수를 호출합니다.
        // 반환형은 std::vector<std::vector<Detection>>입니다.
        auto batchDetections = detector.detectBatchPreprocessed(batchBuffers, origSizes, 0.25f, 0.45f);

        for (size_t i = 0; i < batch.size(); ++i) {
            for (const auto& det : batchDetections[i]) {
                cv::rectangle(*batch[i].imagePtr, det.box, cv::Scalar(0, 255, 0), 2);
            }
        }

        totalProcessedFrames += static_cast<int>(batch.size());
        totalBatches++;
    }
}

int main() {
    std::string engineName = "yolov8n.dynamic_b4.int8.engine";
    YoloTRT detector(engineName);
    std::cout << "사용한 엔진: " << engineName << std::endl;

    // 테스트용으로 같은 비디오를 4개 사용합니다. 카메라도 4대라고 가정하는 겁니다.
    std::vector<std::string> videoPaths = {
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4"
    };

    const size_t maxBatchSize = 4;
    const int timeoutMs = 15; // 15ms 동안 배치에 4개가 채워지지 않으면 지체하지 않고 태워서 추론으로 보냅니다.

    // Warm up 합니다.
    std::vector<cv::Mat> dummyBatch(maxBatchSize, cv::Mat::zeros(640, 640, CV_8UC3));
    for (int i = 0; i < 30; ++i) {
        detector.detectBatch(dummyBatch, 0.25f, 0.45f);
    }

    DynamicBatchCollector collector(maxBatchSize, timeoutMs);
    std::atomic<bool> stopSignal{false};
    int totalProcessedFrames = 0;
    int totalBatches = 0;

    auto totalStart = std::chrono::high_resolution_clock::now();

    // 추론 스레드 시작합니다. consumer
    std::thread inferencer(
        batchInferenceThread, 
        std::ref(detector), 
        std::ref(collector), 
        std::ref(stopSignal), 
        std::ref(totalProcessedFrames), 
        std::ref(totalBatches), 
        maxBatchSize
    );

    // 총 4개의 카메라 스레드를 시작합니다. producer
    std::vector<std::thread> cameraThreads;
    for (int i = 0; i < static_cast<int>(videoPaths.size()); ++i) {
        cameraThreads.emplace_back(cameraStreamThread, i, videoPaths[i], std::ref(collector), std::ref(stopSignal));
    }

    // 모든 카메라 스트림이 끝날 때까지 대기합니다.
    for (auto& t : cameraThreads) {
        if (t.joinable()) t.join();
    }

    // 모든 카메라 종료 후 추론 스레드를 깨워서 종료시킵니다.
    stopSignal = true;
    collector.notifyStop();
    if (inferencer.joinable()) inferencer.join();

    auto totalEnd = std::chrono::high_resolution_clock::now();
    double totalSec = std::chrono::duration<double>(totalEnd - totalStart).count();

    std::cout << "총 처리 프레임 수: " << totalProcessedFrames << " 프레임" << std::endl;
    std::cout << "총 배치 횟수: " << totalBatches << "회 (평균 배치 크기: "
              << cv::format("%.2f", static_cast<double>(totalProcessedFrames) / totalBatches) << "장)" << std::endl;
    std::cout << "총 소요 시간: " << cv::format("%.2f", totalSec) << "초" << std::endl;
    std::cout << "평균 FPS: " << cv::format("%.1f", totalProcessedFrames / totalSec) << " FPS" << std::endl;
}
```

```cpp
#include "YoloTRT_orig.h"
#include <iostream>
#include <vector>
#include <queue>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <chrono>
#include <memory>
#include <atomic>
#include <opencv2/opencv.hpp>

// 흐름 정리
/**

*/

// 실제 이미지(cv::Mat)의 포인터가 담긴 구조체입니다.
struct FramePacket {
    int streamId; // 카메라 ID
    int frameNumber; // 프레임 번호(ex: 1 ~ 3129)
    cv::Mat image; // 카메라 프레임이 덮어씌워질 버퍼
    std::vector<float> preprocessedData; // 전처리된 CHW 데이터가 덮어씌워질 버퍼
    cv::Size origSize;

    // 생성 시점에 단 한 번만 4.91MB 메모리를 확보해 둡니다.
    FramePacket() : preprocessedData(YoloTRT::kSingleInputElements) {}
};

// 여러 카메라 스레드와 추론 스레드가 안전하게 패킷을 빌리고 반납하는 스레드 안전 메모리 풀
class PacketObjectPool {
private:
    std::queue<FramePacket*> pool_;
    std::vector<std::unique_ptr<FramePacket>> allPackets_; // 전체 패킷의 실제 수명 관리자
    std::mutex mtx_;
    std::condition_variable cvAvailable_;
public:
    // poolSize 개수만큼 패킷을 미리 생성해 풀에 채워 둡니다.
    explicit PacketObjectPool(size_t poolSize) {
        allPackets_.reserve(poolSize);
        for (size_t i = 0; i < poolSize; ++i) {
            auto pkt = std::make_unique<FramePacket>();
            pool_.push(pkt.get());
            allPackets_.push_back(std::move(pkt));
        }
    }

    std::shared_ptr<FramePacket> acquire(const std::atomic<bool>& stopSignal) {
        std::unique_lock<std::mutex> lock(mtx_);

        cvAvailable_.wait(lock, [this, &stopSignal]() {
            return !pool_.empty() || stopSignal.load();
        });

        if (pool_.empty() && stopSignal.load()) return nullptr;

        FramePacket* rawPtr = pool_.front();
        pool_.pop();

        // 스마트 포인터의 참조 카운트가 0이 될 때 delete 대신 풀로 반납하는 커스텀 삭제자
        return std::shared_ptr<FramePacket>(rawPtr, [this](FramePacket* p) {
            std::lock_guard<std::mutex> returnLock(mtx_);
            pool_.push(p);
            cvAvailable_.notify_one(); // 반납되었음을 대기 중인 카메라 스레드에 알립니다.
        });
    }
    
    void notifyStop() {
        std::lock_guard<std::mutex> lock(mtx_);
        cvAvailable_.notify_all();
    }
};

// 배치를 모으고(push), 추론하는(popBatch) 클래스입니다.
class DynamicBatchCollector {
private:
    std::vector<std::shared_ptr<FramePacket>> buffer_; // 프레임 패킷을 배치 단위로 모을 벡터입니다.
    size_t maxBatchSize_; // 배치의 최대 개수입니다. 여기서는 4로 했습니다.
    std::chrono::milliseconds timeout_; // 한 배치에 4개가 쌓이지 않아도 일정 시간이 지나면 태워서 추론에 보내도록 타임아웃 시간을 설정합니다.
    std::mutex mtx_; // 여러 스레드의 동시 접근을 막는 mutex입니다.

    // 버퍼(buffer_)에 데이터가 들어올 때까지 추론 스레드를 재우고 깨우는 조건변수입니다.
    std::condition_variable cvNotEmpty_;
    // 버퍼에 빈 공간이 생길 때까지 카메라 스레드를 재우는 조건변수입니다.
    std::condition_variable cvNotFull_;
public:
    // 생성자: 멤버 초기화 리스트(:)를 사용해 변수들을 초기화합니다.
    // 불필요한 기본 생성 후 대입 과정을 생략해서 성능을 높입니다.
    DynamicBatchCollector(size_t maxBatchSize, int timeoutMs)
        : maxBatchSize_(maxBatchSize), timeout_(std::chrono::milliseconds(timeoutMs)) {
        buffer_.reserve(maxBatchSize_); // 동적 할당 방지를 위해 메모리 공간을 미리 예약합니다.
    }

    // 프레임 패킷을 우측값 참조(&&)로 받습니다. 소유권 이동(std::move)를 하기 위해서입니다.
    bool push(std::shared_ptr<FramePacket> packet, const std::atomic<bool>& stopSignal) {
        std::unique_lock<std::mutex> lock(mtx_); // 다른 스레드가 접근하지 못하게 자물쇠를 걸어 잠급니다.

        // 버퍼(buffer_)가 가득차지 않았다면(공간이 있다면) 밑으로 통과하게 합니다.
        // 공간이 없다면, 공간이 생길 때까지 스레드를 재워서 CPU 낭비를 막습니다.
        cvNotFull_.wait(lock, [this, &stopSignal]() {
            return buffer_.size() < maxBatchSize_ || stopSignal.load();
        });

        if (stopSignal.load()) return false;

        buffer_.push_back(std::move(packet)); // 소유권 이동함으로써, 복사비용 없이 버퍼(buffer_)에 적재합니다.
        cvNotEmpty_.notify_one(); // 버퍼에 프레임 패킷이 들어왔으니 추론 스레드를 깨웁니다.
        return true;
    }

    // 바꿔치기(swap)하기 위해 빈 벡터를 가져옵니다.
    bool popBatch(std::vector<std::shared_ptr<FramePacket>>& outBatch, const std::atomic<bool>& stopSignal) {
        std::unique_lock<std::mutex> lock(mtx_); // 다른 스레드가 접근하지 못하게 자물쇠를 걸어 잠급니다.

        // 버퍼(buffer_)가 비어있지 않다면(프레임이 존재한다면) 밑으로 통과하게 합니다.
        // 프레임이 존재하지 않는다면 생길 때까지 스레드를 재워서 CPU 낭비를 막습니다.
        cvNotEmpty_.wait(lock, [this, &stopSignal]() {
            return !buffer_.empty() || stopSignal.load();
        });

        // 버퍼(buffer_)가 비어있고, 종료신호가 켜졌다면 false를 반환합니다.
        if (buffer_.empty() && stopSignal.load()) return false;

        // 버퍼(buffer_) 안에 4개 이상(4개를 초과할 수 없기 때문에 사실상 4개가 맥시멈)이거나 지정한 시간(timeout_)이 지났을 때 밑으로 통과하게 합니다.
        cvNotEmpty_.wait_for(lock, timeout_, [this, &stopSignal]() {
            return buffer_.size() >= maxBatchSize_ || stopSignal.load();
        });

        // 버퍼(buffer_)에 모인 프레임들의 소유권을 통째로 outBatch로 이동(swap)시킵니다.
        outBatch.swap(buffer_);

        // 버퍼를 완전히 비웠으니 대기 중이던 카메라 스레드들을 모두 깨웁니다.
        cvNotFull_.notify_all();
        return true;
    }

    void notifyStop() {
        std::lock_guard<std::mutex> lock(mtx_);
        cvNotEmpty_.notify_all();
        cvNotFull_.notify_all();
    }
};

// 카메라 4대가 쓰는 스레드입니다. VideoCapture 객체에서 나온 프레임을 프레임 패킷 구조체에 담아서 push합니다.
void cameraStreamThread(
    int streamId, 
    const std::string& videoPath, 
    PacketObjectPool& pool, 
    DynamicBatchCollector& collector, 
    const std::atomic<bool>& stopSignal
) {
    cv::VideoCapture cap(videoPath);
    if (!cap.isOpened()) {
        std::cerr << "[카메라 " << streamId << "] 비디오 열기 실패\n";
        return;
    }

    int frameIdx = 0;
    while (!stopSignal.load()) {
        // new / make_shared 없이 풀에서 이미 할당된 패킷을 빌려옵니다.
        auto packet = pool.acquire(stopSignal);
        if (!packet) break;

        // 기존 packet->image 버퍼 위에 새 프레임을 그대로 덮어씁니다. 메모리 재할당 없음
        cap >> packet->image;
        if (packet->image.empty()) break;

        frameIdx++;
        packet->streamId = streamId;
        packet->frameNumber = frameIdx;
        packet->origSize = packet->image.size();

        // 이미 4.91MB가 확보된 packet->preprocessedData 위에 전처리 결과만 덮어씁니다.
        YoloTRT::preprocessSingle(packet->image, packet->preprocessedData);

        if (!collector.push(std::move(packet), stopSignal)) break;
    }
    cap.release();
}

// 배치 단위로 들어온 프레임을 consumer가 추론하는 스레드입니다.
void batchInferenceThread(
    YoloTRT& detector, 
    DynamicBatchCollector& collector, 
    const std::atomic<bool>& stopSignal, 
    int& totalProcessedFrames, 
    int& totalBatches, 
    const size_t maxBatchSize
) {
    std::vector<std::shared_ptr<FramePacket>> batch;
    std::vector<std::vector<float>> batchBuffers;
    // 후처리할 때 원본 크기로 복원이 필요하므로, cv::Size를 담을 벡터를 하나 선언합니다.
    std::vector<cv::Size> origSizes;

    batch.reserve(maxBatchSize);
    batchBuffers.reserve(maxBatchSize);
    origSizes.reserve(maxBatchSize);

    while (true) {
        // 이전 배치의 batch.clear()가 실행되는 순간, 
        // 각 shared_ptr의 커스텀 삭제자가 호출되어 패킷들이 자동으로 Object Pool에 반환됨
        batch.clear();
        batchBuffers.clear();
        origSizes.clear();

        if (!collector.popBatch(batch, stopSignal)) break;

        for (auto& pkt : batch) {
            batchBuffers.push_back(pkt->preprocessedData);
            origSizes.push_back(pkt->origSize);
        }

        // 전처리 과정이 생략된 배치 추론 함수를 호출합니다.
        // 반환형은 std::vector<std::vector<Detection>>입니다.
        auto batchDetections = detector.detectBatchPreprocessed(batchBuffers, origSizes, 0.25f, 0.45f);


        for (size_t i = 0; i < batch.size(); ++i) {
            for (const auto& det : batchDetections[i]) {
                cv::rectangle(batch[i]->image, det.box, cv::Scalar(0, 255, 0), 2);
            }
        }

        totalProcessedFrames += static_cast<int>(batch.size());
        totalBatches++;
    }
}

int main() {
    std::string engineName = "yolov8n.dynamic_b4.int8.engine";
    YoloTRT detector(engineName);
    std::cout << "사용한 엔진: " << engineName << std::endl;

    // 테스트용으로 같은 비디오를 4개 사용합니다. 카메라도 4대라고 가정하는 겁니다.
    std::vector<std::string> videoPaths = {
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4", 
        "./sample_data/sample.mp4"
    };

    const size_t maxBatchSize = 4;
    const int timeoutMs = 15; // 15ms 동안 배치에 4개가 채워지지 않으면 지체하지 않고 태워서 추론으로 보냅니다.
    const size_t poolSize = 16;

    // Warm up 합니다.
    std::vector<cv::Mat> dummyBatch(maxBatchSize, cv::Mat::zeros(640, 640, CV_8UC3));
    for (int i = 0; i < 30; ++i) {
        detector.detectBatch(dummyBatch, 0.25f, 0.45f);
    }

    PacketObjectPool pool(poolSize);
    DynamicBatchCollector collector(maxBatchSize, timeoutMs);
    std::atomic<bool> stopSignal{false};
    int totalProcessedFrames = 0;
    int totalBatches = 0;

    auto totalStart = std::chrono::high_resolution_clock::now();

    // 추론 스레드 시작합니다. consumer
    std::thread inferencer(
        batchInferenceThread, 
        std::ref(detector), 
        std::ref(collector), 
        std::ref(stopSignal), 
        std::ref(totalProcessedFrames), 
        std::ref(totalBatches), 
        maxBatchSize
    );

    // 총 4개의 카메라 스레드를 시작합니다. producer
    std::vector<std::thread> cameraThreads;
    for (int i = 0; i < static_cast<int>(videoPaths.size()); ++i) {
        cameraThreads.emplace_back(cameraStreamThread, 
            i, 
            videoPaths[i], 
            std::ref(pool), 
            std::ref(collector), 
            std::ref(stopSignal)
        );
    }

    // 모든 카메라 스트림이 끝날 때까지 대기합니다.
    for (auto& t : cameraThreads) {
        if (t.joinable()) t.join();
    }

    // 모든 카메라 종료 후 추론 스레드를 깨워서 종료시킵니다.
    stopSignal = true;
    pool.notifyStop();
    collector.notifyStop();
    if (inferencer.joinable()) inferencer.join();

    auto totalEnd = std::chrono::high_resolution_clock::now();
    double totalSec = std::chrono::duration<double>(totalEnd - totalStart).count();

    std::cout << "총 처리 프레임 수: " << totalProcessedFrames << " 프레임" << std::endl;
    std::cout << "총 배치 횟수: " << totalBatches << "회 (평균 배치 크기: "
              << cv::format("%.2f", static_cast<double>(totalProcessedFrames) / totalBatches) << "장)" << std::endl;
    std::cout << "총 소요 시간: " << cv::format("%.2f", totalSec) << "초" << std::endl;
    std::cout << "평균 FPS: " << cv::format("%.1f", totalProcessedFrames / totalSec) << " FPS" << std::endl;
}
```
