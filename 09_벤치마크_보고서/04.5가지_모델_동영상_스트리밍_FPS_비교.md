# YOLOv8 동영상 스트리밍 처리 속도 비교 (PyTorch vs ONNX Runtime vs TensorRT)

YOLOv8n 모델의 동영상 스트리밍 처리 속도를 벤치마킹한 결과입니다.

제 컴퓨터는 M4 맥미니라서 TensorRT 환경 구축이 안되기 때문에 구글 코랩의 T4 GPU를 사용하였습니다.

동일한 조건에서 1000회 추론을 했을 때 평균, 최소, 최대 지연 시간 및 FPS를 측정했습니다.

한 프레임당 평균 소요 시간은 9.49 ms입니다. FPS로 따지면 105.3 FPS입니다.

특히 PyTorch는 Max Latency가 135.35 ms를 찍는 반면 C++ TensorRT는 Max Latency가 21.76 ms에 불과하여 아주 안정적입니다.

이렇게 차이가 나는 이유는 C++은 파이썬에 비해 GIL(Global Interpreter Lock)이 없어서 병렬처리가 가능합니다.

그리고 파이썬은 인터프리터가 매 라인마다 객체를 래핑하고 타입을 변환하고 C++ 레이어로 올려주는 과정의 오버헤드가 있는데, C++은 그러한 오버헤드가 없기 때문에 엄청난 속도를 보여줍니다.

```python
import torch
from ultralytics import YOLO
import cv2
import time
import numpy as np

# 모델 로드 및 CUDA 최적화 설정
# 동영상 추론은 프레임마다 동일한 크기(640 640)로 들어옵니다. 이처럼 입력 크기가 고정된 작업에서는 CuDNN 라이브러리가
# 현재 GPU에 가장 최적화된 알고리즘을 탐색(benchmark)해서 고정하도록 만들어 줍니다.
# 동일한 크기의 프레임을 초고속으로 처리할 수 있기 때문에 동영상 스트리밍 처리에 필수로 들어가는 옵션입니다.
torch.backends.cudnn.benchmark = True
model = YOLO('yolov8n.pt')

# OpenCV를 이용해서 비디오 파일을 엽니다.
video_path = './test_videos/sample.mp4'
cap = cv2.VideoCapture(video_path)

if not cap.isOpened():
    raise FileNotFoundError(f'비디오 파일을 열 수 없습니다: {video_path}')

# 비디오 파일의 총 프레임 개수를 반환합니다.
total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))

# Warm-Up
ret, frame = cap.read()
if ret:
    with torch.no_grad():
	for _ in range(50):
	    _ = model(frame, verbose=False)
    torch.cuda.synchronize()

# 비디오 캡처 위치를 맨 처음(0번 프레임)으로 리셋합니다.
cap.set(cv2.CAP_PROP_POS_FRAMES, 0)

# 동영상 프레임 측정하기 전에 소요시간을 담을 latencies, 프레임 개수를 셀 frame_count를 선언합니다.
latencies = []
frame_count = 0

# 역전파를 금지해서 추론을 가속화합니다.
with torch.no_grad():
    while cap.isOpened():
        # ret은 true or false를 반환합니다. 다음 프레임이 있다면 true. 그렇지 않다면 false이므로, false일 때는 반복문을 종료합니다.
	ret, frame = cap.read()
	if not ret:
	    break

	start = time.time()
        # model() 함수 안에 전처리 + 순수 추론 + 후처리 작업이 모두 포함되어 있습니다.
	results = model(frame, verbose=False)
        # plot() 함수를 써서 바운딩박스 처리가 된 이미지를 바로 반환합니다.
	annotated_frame = results[0].plot()

	torch.cuda.synchronize()

	latencies.append((time.time() - start) * 1000)
	frame_count += 1

# 반복문이 끝났으므로 VideoCapture 객체를 해제하고 자원을 반납합니다.
cap.release()

avg = np.mean(latencies)

print('================ [1. PyTorch Video End-to-End 벤치마크] ================')
print(f'처리 프레임 수: {frame_count} / {total_frames}')
print(f'Avg Latency: {avg:.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print(f'Avg FPS: {1000.0 / avg:.1f} FPS')
print('==========================================================================')
```

<summary><b>2. ONNX Runtime 측정 코드 및 결과</b></summary>

```Python
import onnxruntime as ort
import cv2
import numpy as np
import time

# ONNX 추론 세션을 생성합니다.
# 1순위로 GPU(CUDA)를 시도합니다. GPU와 드라이버가 잘 갖춰져 있다면 바로 GPU 가속으로 동작합니다.
# 하지만 잘 갖춰져 있지 않다면 2순위로 CPU를 시도합니다. GPU보다 느리지만.. 어쩔 수 없습니다.
providers = ['CUDAExecutionProvider', 'CPUExecutionProvider']
session = ort.InferenceSession('yolov8n.onnx', providers=providers)
input_name = session.get_inputs()[0].name
output_name = session.get_outputs()[0].name

# 전처리 함수입니다.
# 640 640으로 리사이즈하고, BGR을 RGB로 변환하고,
# HWC(640, 640, 3)을 CHW(3, 640, 640)으로 변환하고 0과 1 사이의 실수값으로 정규화하고 차원을 추가(expand_dims)하여
# (1, 3, 640, 640)으로 shape을 만들어서, 모델의 입력으로 들어갈 준비를 마칩니다.
def preprocess(frame):
    img_resized = cv2.resize(frame, (640, 640))
    img_rgb = cv2.cvtColor(img_resized, cv2.COLOR_BGR2RGB)
    tensor = img_rgb.transpose(2, 0, 1).astype(np.float32) / 255.0
    return np.expand_dims(tensor, axis=0)

# 후처리 함수입니다.
# outputs[0]의 shape은 (1, 84, 8400)이므로 연산을 편하게 하기 위해 squeeze를 해서 1을 제거합니다.
# boxes의 shaped은.. 앞 4개(cx, cy, boxW, boxH)만 가져오므로 (4, 8400)입니다.
# scores: 뒤의 모든 것을 다 가져오면 (80, 8400)인데, 여기서 행(axis=0)[세로방향]을 기준으로 8400개 박스 각각에 대해 max를 써서 가장 확신도가 높은 score를 채택을 합니다. 그러면 scores의 shape은 (8400,)이 됩니다.
# class_ids: argmax를 써서 가장 높은 score를 가진 클래스 아이디를 가져옵니다. 이 역시 shape은 (8400,)이 됩니다.
# valid_mask: 8400개의 score가 확신도 임계값(conf_thresh)를 넘으면 true, 그렇지 않으면 false이므로 이 역시 (8400,)입니다.
def postprocess(outputs, orig_shape, conf_thresh=0.25):
    preds = np.squeeze(outputs[0]) # (84, 8400)
    boxes = preds[:4, :]
    scores = np.max(preds[4:, :], axis=0)
    class_ids = np.argmax(preds[4:, :], axis=0)

    valid_mask = scores > conf_thresh
    # 임계값을 넘는 score 개수가 a라면, boxes의 shape은 (4, a). scores의 shape은 (a,). class_ids의 shape은 (a,)가 됩니다.
    boxes = boxes[:, valid_mask]
    scores = scores[valid_mask]
    class_ids = class_ids[valid_mask]

    # 전처리 과정에서 원본 이미지를 640 640으로 리사이즈 했으므로, 탐지된 박스도 리사이즈된 사이즈 기준입니다.
    # 이것을 원본으로 복원하는 과정이 필요하기 때문에,
    # 원본 이미지의 가로(orig_w), 세로(orig_h) 길이를 각각 640으로 나눈 값을 각각 scale_x, scale_y로 저장합니다.
    # 나중에 rectangle을 하기 전의 박스의 크기를 보정할 때 이 값을 곱해주어야 하기 때문에 필요합니다.
    orig_h, orig_w = orig_shape[:2]
    scale_x = orig_w / 640.0
    scale_y = orig_h / 640.0

    boxes_rescaled = [] # 원본 크기로 복원된 네모박스를 담기 위해 배열을 선언합니다.
    for box in boxes.T: # boxes.T의 shape은 (a, 4)가 됩니다. a개의 네모박스가 있는 것입니다.
        cx, cy, w, h = box
        # 기존 방식으로 xmin, ymin, bw, bh를 구한 뒤에 원본 크기로 복원하기 위해 scale_x, scale_y 값을 곱해줍니다.
        xmin = int((cx - w / 2) * scale_x)
        ymin = int((cy - h / 2) * scale_y)
        bw = int(w * scale_x)
        bh = int(h * scale_y)
        # 복원된 xmin, ymin, bw, bh 값을 배열에 담아줍니다.
        boxes_rescaled.append([xmin, ymin, bw, bh])
    # 배열, 점수(scores), 클래스 아이디를 반환합니다.
    return boxes_rescaled, scores, class_ids

# OpenCV를 이용해서 비디오 파일을 엽니다.
video_path = './test_videos/sample.mp4'
cap = cv2.VideoCapture(video_path)

if not cap.isOpened():
    raise FileNotFoundError(f'비디오 파일을 열 수 없습니다: {video_path}')

# 비디오의 총 프레임 개수를 가져옵니다.
total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))

# Warm-Up
ret, frame = cap.read()
if ret:
    for _ in range(50):
	input = preprocess(frame)
	output = session.run([output_name], {input_name: input})
	_ = postprocess(output, frame.shape)

cap.set(cv2.CAP_PROP_POS_FRAMES, 0)

# 동영상 프레임 측정하기 전에 프레임 처리시간을 담을 latencies 배열, 프레임 개수를 셀 frame_count를 선언합니다.
latencies = []
frame_count = 0

while cap.isOpened():
    ret, frame = cap.read()
    if not ret:
        break

    start = time.time()

    input = preprocess(frame)
    output = session.run([output_name], {input_name: input})
    # 후처리해서 나온 최종 후보 박스들(boxes), 점수들(scores), class_ids를 가져옵니다.
    boxes, scores, class_ids = postprocess(output, frame.shape)

    # 네모박스 정보를 가지고서 rectangle 함수를 사용해서 실제로 프레임(frame) 위에 네모박스를 그립니다.
    # 그 네모박스가 어떤 클래스를 의미하고, 그 클래스의 확신도가 몇인지 putText 함수를 통해 보여줍니다.
    for i, box in enumerate(boxes):
        xmin, ymin, bw, bh = box
        cv2.rectangle(frame, (xmin, ymin), (xmin + bw, ymin + bh), (0, 255, 0), 2)
        text = f'class_id: {class_ids[i]}, confidence: {scores[i]:.2f}'
        cv2.putText(frame, text, (xmin, ymin - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 0, 0), 2)

    latencies.append((time.time() - start) * 1000)
    frame_count += 1

# 반복문이 끝났으므로 VideoCapture 객체를 해제하고 자원을 반납합니다.
cap.release()

avg = np.mean(latencies)

print('================ [2. ONNX Runtime Video End-to-End 벤치마크] ================')
print(f'처리 프레임 수: {frame_count} / {total_frames}')
print(f'Avg Latency: {avg:.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print(f'Avg FPS: {1000.0 / avg:.1f} FPS')
print('==========================================================================')
```

<summary><b>TensorRT 가속 엔진 (.engine) 변환 코드</b></summary>

```python
import os
import tensorrt as trt

# Logger와 Builder를 생성합니다.
TRT_LOGGER = trt.Logger(trt.Logger.WARNING)
builder = trt.Builder(TRT_LOGGER)

# 네트워크: 모델의 레이어와 구조를 담을 빈 설계도입니다.
# 아직 아무것도 없는 상태이고, 여기에 ONNX 모델의 레이어와 구조를 채워넣을 것입니다.
network = builder.create_network()
parser = trt.OnnxParser(network, TRT_LOGGER)

onnx_path = 'yolov8n.onnx'
if not os.path.exists(onnx_path):
    raise FileNotFoundError(f'{onnx_path} 파일이 없습니다.')

# 바이너리 형태로 ONNX 파일을 읽어낸 뒤에, 파싱합니다.
with open(onnx_path, 'rb') as model:
    if not parser.parse(model.read()):
	print('ONNX 파싱 실패')
	raise RuntimeError('ONNX 파서 에러')

# config: 이 모델을 굽는데 필요한 설정(최적화를 위한 옵션)입니다.
config = builder.create_builder_config()

# 최적화 설정을 바탕으로 설계도(network)를 구워서 직렬화된 바이트 데이터로 만듭니다.
serialized_engine = builder.build_serialized_network(network, config)

if serialized_engine is None:
    raise RuntimeError('가속 엔진 빌드 프로세스 실패')

# 읽은 바이너리 데이터를 write binary하여 엔진 파일을 생성합니다.
engine_path = 'yolov8n.engine'
with open(engine_path, 'wb') as f:
    f.write(serialized_engine)

# 만들어진 engine 파일을 읽기 위해 Runtime 객체를 생성합니다.
runtime = trt.Runtime(TRT_LOGGER)

# 비직렬화 합니다.
with open(engine_path, 'rb') as f:
    engine = runtime.deserialize_cuda_engine(f.read())

# 엔진을 바탕으로 실제 추론을 실행할 context를 준비합니다.
context = engine.create_execution_context()
```

```Python
input_shape = (1, 3, 640, 640)
input_size = trt.volume(input_shape) * np.dtype(np.float32).itemsize
output_shape = tuple(engine.get_tensor_shape(engine.get_tensor_name(1)))
output_size = trt.volume(output_shape) * np.dtype(np.float32).itemsize

# input_size, output_size만큼 메모리를 동적 할당합니다.
d_input = cuda.mem_alloc(input_size)
d_output = cuda.mem_alloc(output_size)

# CPU에서 GPU로 작업을 던질 통로(Stream)를 생성합니다.
stream = cuda.Stream()

# Host(CPU)에서 출력결과를 받아낼 그릇, output_shape 크기의 넘파이 배열을 만듭니다.
host_output = np.empty(output_shape, dtype=np.float32)

# 입력, 출력 포인터에 입출력 이름을 매핑해 줍니다.
context.set_tensor_address(engine.get_tensor_name(0), int(d_input))
context.set_tensor_address(engine.get_tensor_name(1), int(d_output))

# 이미지(프레임)를 매개변수로 받는 전처리 함수입니다.
def preprocess(frame):
    img_resized = cv2.resize(frame, (640, 640)) # 640 640으로 리사이즈합니다.
    img_rgb = cv2.cvtColor(img_resized, cv2.COLOR_BGR2RGB) # BGR을 RGB로 변환합니다.
    # transpose() 함수를 써서 shape을 (3, 640, 640)으로 바꿉니다.
    # 데이터 타입을 실수(float32)로 바꾸고, 255로 나눠서 0과 1 사이의 실수 값을 가지게 하는 정규화 작업을 합니다.
    tensor = img_rgb.transpose(2, 0, 1).astype(np.float32) / 255.0
    # GPU 메모리에 데이터를 한 번에 복사하기 위해, 데이터를 메모리상에 빈틈없이 한줄로 나란히 합니다.
    return np.ascontiguousarray(np.expand_dims(tensor, axis=0))

# 탐지 결과 preds( (1, 84, 8400) shape ), 원래 이미지의 shape인 orig_shape, 확신도 임계값 conf_thresh를 매개변수로 받는 후처리 함수입니다.
def postprocess(preds, orig_shape, conf_thresh=0.25):
    boxes = preds[0, :4, :] # 앞 4개(:4)를 가져오므로 shape은 (4, 8400)입니다.
    # 84개 중 4개를 제외한 뒤의 전부(4:)를 가져오므로 shape은 (80, 8400)인데,
    # np.max() 함수. axis=0(세로 방향)을 기준으로 최댓값을 가져오니까 최종 shape은 (8400,)이 됩니다.
    scores = np.max(preds[0, 4:, :], axis=0)
    # np.argmax() 함수. axis=0(세로 방향)을 기준으로 최댓값이 위치한 위치의 인덱스 값. 클래스 번호를 가져옵니다. shape은 (8400,)
    class_ids = np.argmax(preds[0, 4:, :], axis=0)
    # scores에 있는 각각의 score가 확신도 임계값(conf_thresh)를 넘으면 true, 넘지 못하면 false를 가지므로 shape은 동일하게 (8400,)
    valid_mask = scores > conf_thresh # (8400,)

    # boxes에 있는 모든 8400개의 박스 중에서 true인 것들만 가져옵니다. 그 개수가 a라면, shape은 (4, a)가 됩니다.
    valid_boxes = boxes[:, valid_mask]
    # scores에 있는 모든 8400개의 점수 중에서 true인 것들만 가져옵니다. shape은 (a,)가 됩니다.
    valid_scores = scores[valid_mask]

    for box in valid_boxes.T:
```

```Python
# 그렇지 않으면 에러를 뱉습니다.
output_shape = tuple( engine.get_tensor_shape(engine.get_tensor_name(1)) )

# pagelocked_empty의 의미
# CPU 메모리를 할당할 때 메모리스왑하지 말고 RAM에 고정(Pinned)시켜달라고 요청합니다.
h_input = cuda.pagelocked_empty(input_shape, dtype=np.float32)
h_output = cuda.pagelocked_empty(output_shape, dtype=np.float32)

# h_input, h_output의 크기만큼 메모리를 미리 할당합니다.
d_input = cuda.mem_alloc(h_input.nbytes)
d_output = cuda.mem_alloc(h_output.nbytes)

stream = cuda.Stream()

# set_tensor_address(name, ptr)
# 추론을 수행할 때 입력 및 출력 텐서의 GPU 메모리 버퍼 주소(포인터)를 지정하는 메서드입니다.

context.set_tensor_address(engine.get_tensor_name(0), int(d_input))
context.set_tensor_address(engine.get_tensor_name(1), int(d_output))

# 고정된 메모리 공간(pinned_buf)에 입력 데이터(frame)를 복사해서 집어넣는 과정이 필요합니다.

# 그냥 pinned_buf = np.expand_dims(tensor, axis=0)을 해버리면.. 메모리 주소가 새로 할당되면서 Pinned memory 특징이 사라질 수 있습니다.
def preprocess_to_pinned(frame, pinned_buf):
    img_resized = cv2.resize(frame, (640, 640)) # shape: (640, 640, 3)
    img_rgb = cv2.cvtColor(img_resized, cv2.COLOR_BGR2RGB)
    tensor = img_rgb.transpose(2, 0, 1).astype(np.float32) / 255.0 # shape: (3, 640, 640)
    np.copyto(pinned_buf, np.expand_dims(tensor, axis=0))

# 탐지 결과인 preds(shape은 (1, 84, 8400)), 원래 이미지의 shape인 orig_shape, 확신도 임계값 conf_thresh를 매개변수로 받는 후처리 함수입니다.
def postprocess(preds, orig_shape, conf_thresh=0.25):
```

### `YoloTRT.cpp`
```cpp
#include "YoloTRT.h"

YoloTRT::YoloTRT(const std::string& enginePath) {
    std::ifstream file(enginePath, std::ios::binary);
    if (!file.good()) {
        std::cerr << "엔진 파일을 찾을 수 없습니다: " << enginePath << std::endl;
        return; 
    }
    file.seekg(0, file.end); // file 커서를 맨 끝(end)으로 이동합니다.
    size_t size = file.tellg(); // 현재 커서의 위치(end)는 파일의 전체 바이트 크기를 의미하므로, 이것을 size_t 변수로 담습니다.
    file.seekg(0, file.beg); // file 커서를 다시 맨 처음(beg)으로 이동합니다.

    std::vector<char> engineData(size); // file 크기만큼의 빈 공간을 가진 1차원 벡터를 생성합니다.
    // 벡터가 메모리에 할당된 시작 주소(engineData.data())를 통해,
    // file에 들어있던 바이너리 데이터를 그 시작 주소로부터 size 만큼 통째로 읽어와서 복사합니다.
    file.read(engineData.data(), size);
    file.close();

    runtime = nvinfer1::createInferRuntime(gLogger);
    // engineData에는 .engine 파일의 바이너리가 1차원 벡터로 담겨 있는데, .data()를 써서 그 1차원 주소의 시작값을 넘겨주고,
    // 주소 시작값부터 size만큼의 데이터를 역직렬화해서 엔진을 생성합니다.
    engine = runtime->deserializeCudaEngine(engineData.data(), size);
    context = engine->createExecutionContext();

    // CPU에서 GPU로 데이터를 전송할 통로(Stream)를 생성합니다.
    cudaStreamCreate(&stream);

    // inputSize, outputSize만큼의 메모리를 미리 동적 할당합니다.
    // 그리고 할당된 메모리의 시작 주소를 d_input, d_output에 담아줍니다.
    // 여기서 &로 던져주는 이유.
    // 그냥 d_input으로 던져주게 되면 그냥 값만 넘겨주는거지, 주소를 넘겨주진 못하기 때문에 d_input에는 아무런 변화가 없습니다.
    // 그래서 &d_input으로 넘겨줘야, 문신처럼 각인할 수 있습니다.
    cudaMalloc(&d_input, inputSize);
    cudaMalloc(&d_output, outputSize);

    // cudaHostAlloc(): CUDA에서 Pinned Memory를 Host(CPU) 측에 할당하는 함수입니다.
    // 일반적으로 malloc()을 통해 할당한 메모리는 OS에 의해 디스크로 페이징(메모리스왑)이 될 수 있는 Pageable memory입니다.
    // cudaHostAlloc()으로 할당된 메모리는 물리 RAM에 고정될(Pinned) 수 있으므로, GPU가 Host Memory에 빠르게 접근할 수 있습니다.
    // 여기서 &로 넘겨주는 이유도 위와 마찬가지로, 문신처럼 각인하기 위함입니다.
    // 그리고 (void**)는 CUDA 함수와 타입을 맞추기 위한 강제 형변환인데,
    // cudaHostAlloc은 사용자가 메모리에 float을 담을지, int를 담을지 알 수 없으므로 어떤 포인터 주소든 다 받을 수 있도록 (void**)로 강제 형변환해서 보냅니다.
    cudaHostAlloc((void**)&h_input, inputSize, cudaHostAllocDefault);
    cudaHostAlloc((void**)&h_output, outputSize, cudaHostAllocDefault);

    // 각 벡터 안에는 최대 8400개의 탐지된 네모박스의 데이터가 들어올 수 있으므로 미리 메모리 공간을 예약(reserve)합니다.
    boxes.reserve(8400);
    confidences.reserve(8400);
    classIds.reserve(8400);
    indices.reserve(8400);

    const char* inputName = engine->getIOTensorName(0);
    const char* outputName = engine->getIOTensorName(1);
    // 모델의 입력과 출력 데이터가 실제로 저장될 메모리 주소(d_input, d_output)를 매핑해 줍니다.
    context->setTensorAddress(inputName, d_input);
    context->setTensorAddress(outputName, d_output);
}

YoloTRT::~YoloTRT() {
    // 추론이 끝나면, 메모리 누수를 막기 위해 모든 자원을 반납합니다.
    if (stream) cudaStreamDestroy(stream);
    if (d_input) cudaFree(d_input);
    if (d_output) cudaFree(d_output);

    if (h_input) cudaFreeHost(h_input);
    if (h_output) cudaFreeHost(h_output);

    if (context) delete context;
    if (engine) delete engine;
    if (runtime) delete runtime;
}

// 전처리 함수입니다.
void YoloTRT::preprocess(const cv::Mat& srcImg) {
    cv::Mat resizedImg;
    cv::resize(srcImg, resizedImg, cv::Size(640, 640)); // 640 640으로 리사이즈합니다.
    cv::cvtColor(resizedImg, resizedImg, cv::COLOR_BGR2RGB); // BGR을 RGB 순서로 바꿉니다.

    cv::Mat floatImg;
    resizedImg.convertTo(floatImg, CV_32FC3, 1.0 / 255.0); // 0~1의 실수값으로 정규화합니다.

    int channelSize = 640 * 640;
    // reinterpret_cast: 이 메모리 주소의 해석 방식을 강제로 바꿔라는 뜻입니다.
    // cv::Mat 객체(floatImg)의 내부 데이터의 실제 주소는 uchar* 형태입니다.
    // 이 메모리 주소를 float* 형식을 가진 포인터로 취급해서, 앞으로 실수 단위로 값을 꺼내 쓸 수 있도록 해달라고 타입 변환하는 것입니다.
    float* data = reinterpret_cast<float*>(floatImg.data);

    // [R0, G0, B0, R1, G1, B1, ...]를 [R0, R1, ..., G0, G1, ..., B0, B1, ...]로 바꿉니다.
    // 그러면 HWC가 CHW로 바뀝니다. (640, 640, 3) -> (3, 640, 640)
    for (int i = 0; i < channelSize; ++i) {
        h_input[i]                   = data[i * 3 + 0]; // R
        h_input[i + channelSize]     = data[i * 3 + 1]; // G
        h_input[i + channelSize * 2] = data[i * 3 + 2]; // B
    }
}

// 후처리 함수입니다.
std::vector<Detection> YoloTRT::postprocess(int origW, int origH, float confThresh, float nmsThresh) {
    // 이전 프레임 때 사용했던 네 개의 벡터들을 깔끔하게 비워주고, 재사용합니다.
    boxes.clear();
    confidences.clear();
    classIds.clear();
    indices.clear();

    // 전처리 과정에서 원본 이미지를 640 640으로 리사이즈 하는 과정이 있었기에, 탐지된 네모박스도 640 640 이미지 기준입니다.
    // 이것을 원본 사이즈로 복원하는 과정이 필요하기 때문에,
    // 원본 이미지의 가로 세로 길이를 각각 640으로 나눈 뒤에 그 값을 scaleX, scaleY로 저장합니다.
    // 나중에 rectangle하기 전에 이 값을 곱해주어 원본 크기로 복원하는 과정이 필요합니다.
    float scaleX = static_cast<float>(origW) / 640.0f;
    float scaleY = static_cast<float>(origH) / 640.0f;

    // 탐지된 8400개의 박스에 대해 반복문을 실행합니다.
    for (int i = 0; i < 8400; i++) {
        // h_output의 shape은 (1, 84, 8400)이고, 앞 4개(0~3)는 cx, cy, bw, bh를 가지고 있습니다.
        float cx = h_output[0 * 8400 + i];
        float cy = h_output[1 * 8400 + i];
        float w  = h_output[2 * 8400 + i];
        float h  = h_output[3 * 8400 + i];

        // 각 80개의 클래스 중에서 확신도가 가장 높은 클래스 아이디와, 그 클래스의 확신도 값을 가져오는 로직입니다.
        float maxScore = 0.0f;
        int maxClassId = -1;
        // 뒤 80개(4~83)는 클래스 아이디를 의미합니다.
        for (int c = 0; c < 80; c++) {
            float score = h_output[(c + 4) * 8400 + i];
            if (score > maxScore) {
                maxScore = score;
                maxClassId = c;
            }
        }

        // 가장 높은 확신도를 가진 클래스의 확신도가 임계값을 넘었을 때의 박스(그리고 확신도, 클래스 아이디)만 벡터에 담습니다.
        if (maxScore > confThresh) {
            float x1 = (cx - w / 2.0f) * scaleX;
            float y1 = (cy - h / 2.0f) * scaleY;
            float boxW = w * scaleX;
            float boxH = h * scaleY;

            cv::Rect box(x1, y1, boxW, boxH); // 탐지된 네모박스입니다.
            cv::Rect bounds(0, 0, origW, origH); // 원본 이미지입니다.

            // 탐지된 네모박스 구했다면서? 근데 왜 원본 이미지와 교집합을 또 구하는데?
            // YOLOv8 모델이 탐지한 네모박스 좌표가 원본 이미지를 벗어났을 수도 있다.
            // 예를 들어, 원본 이미지가 640 640이라고 가정하고, 탐지된 물체는 x1 값이 480이고, boxW가 170이라고 했을 때,
            // xmin은 480, xmax는 650(480 + 170)이 되어버리면 640을 초과하게 된다.
            // 이렇게 되면 오류가 발생할 수 있기 때문에, 원본 이미지와의 교집합을 구함으로써 보정하는 과정이 필요하다.
            box &= bounds; // 원본 이미지와 탐지된 네모박스의 교집합을 구합니다.

            // 탐지된 네모박스, 그 네모박스의 클래스 아이디, 클래스에 대한 확신도를 각각의 벡터에 담습니다.
            boxes.emplace_back(box);
            confidences.emplace_back(maxScore);
            classIds.emplace_back(maxClassId);
        }
    }

    // 탐지된 박스와 확신도를 담은 벡터, 그리고 확신도 임계값, NMS 임계값을 통해 총 8400개의 박스 중에
    // 임계값을 넘는 박스의 인덱스가 몇번인지 추출합니다.
    // indices에는 만족하는 인덱스 번호가 담깁니다. (예: 54, 74, 5344)
    cv::dnn::NMSBoxes(boxes, confidences, confThresh, nmsThresh, indices);

    // 추출된 인덱스 번호들(indices)만을 가지고 Detection 구조체를 담는 벡터(detections)에 하나씩 밀어넣습니다.
    std::vector<Detection> detections;
    detections.reserve(indices.size());
    for (int idx : indices) {
        // 클래스 생성자에서 reserve로 메모리 공간을 미리 예약해 두었기 때문에 불필요한 메모리 재할당이 발생하지 않습니다!!
        detections.push_back({ boxes[idx], confidences[idx], classIds[idx] });
    }
    // 최종적으로 벡터를 반환합니다.
    return detections;
}

// 실제 추론 함수입니다.
std::vector<Detection> YoloTRT::detect(const cv::Mat& srcImg, float confThresh, float nmsThresh) {
    if (srcImg.empty()) return {};

    // 전처리 과정을 거칩니다.
    preprocess(srcImg);

    // h_input(Host, CPU)에는 이미지의 RGB 정보가 담겨 있고, 그것을 stream을 통해 d_input(Device, GPU)로 inputSize 만큼 복사해 줍니다.
    cudaMemcpyAsync(d_input, h_input, inputSize, cudaMemcpyHostToDevice, stream);
    // 실제 모델 추론입니다.
    context->enqueueV3(stream);
    // 추론을 끝낸 d_output(Device, GPU)에는 (1, 84, 8400)의 데이터가 메모리 공간에 담겨 있고, 그것을 h_output(Host, CPU)로 outputSize만큼 복사해 줍니다.
    cudaMemcpyAsync(h_output, d_output, outputSize, cudaMemcpyDeviceToHost, stream);
    // 정확한 추론 시간을 측정하기 위해 처리가 끝날 때까지 기다려 줍니다.
    cudaStreamSynchronize(stream);
    // 후처리 과정을 진행합니다.
    return postprocess(srcImg.cols, srcImg.rows, confThresh, nmsThresh);
}

void YoloTRT::inferOnly() {
    context->enqueueV3(stream);
    cudaStreamSynchronize(stream);
}
```

### `main.cpp`
```cpp
#include "YoloTRT.h"
#include <chrono>
#include <numeric>
#include <vector>
#include <iostream>
#include <algorithm>
#include <opencv2/opencv.hpp>

int main() {
    // YoloTRT 클래스 객체 생성합니다.
    // 생성자가 호출되며, 엔진 로드 및 입출력 데이터 메모리 동적 할당 등의 추론을 위한 준비 작업을 마칩니다.
    YoloTRT detector("yolov8n.engine");

    // 비디오 파일 준비합니다.
    std::string videoPath = "./test_videos/sample.mp4";
    cv::VideoCapture cap(videoPath);

    if (!cap.isOpened()) {
        std::cerr << "비디오 파일을 열 수 없습니다: " << videoPath << std::endl;
        return -1;
    }

    // 비디오 파일의 총 프레임 수를 가져옵니다.
    int totalFrames = static_cast<int>(cap.get(cv::CAP_PROP_FRAME_COUNT));

    // Warm-up (첫 프레임 기반 50회 예열)
    cv::Mat frame;
    cap >> frame;
    if (!frame.empty()) {
        for (int i = 0; i < 50; ++i) {
            detector.detect(frame, 0.25f, 0.45f);
        }
    }

    // 비디오 읽기 위치를 처음 프레임(0)으로 되돌립니다.
    cap.set(cv::CAP_PROP_POS_FRAMES, 0);

    // 소요시간을 담을 vector를 선언하고, 불필요한 메모리 재할당을 막기 위해 메모리 공간을 미리 예약합니다.
    std::vector<double> latencies;
    latencies.reserve(totalFrames);
    int frameCount = 0;

    // 동영상 처리를 시작합니다.
    while (true) {
        cap >> frame;
        if (frame.empty()) break;

        auto start = std::chrono::high_resolution_clock::now();

        // 추론을 시작합니다.
        auto detections = detector.detect(frame, 0.25f, 0.45f);

        // 탐지된 네모박스를 하나씩 꺼내서 frame에 그립니다.(rectangle)
        for (const auto& det : detections) {
            cv::rectangle(frame, det.box, cv::Scalar(0, 255, 0), 2);
        }

        auto end = std::chrono::high_resolution_clock::now();
        double ms = std::chrono::duration<double, std::milli>(end - start).count();
        latencies.push_back(ms);
        frameCount++;
    }
    // 동영상 처리가 끝났으므로, 객체를 해제합니다.
    cap.release();

    if (latencies.empty()) {
        std::cerr << "처리된 프레임이 없습니다." << std::endl;
        return -1;
    }

    double sum = std::accumulate(latencies.begin(), latencies.end(), 0.0);
    double avg = sum / latencies.size();
    // min과 max를 구하는 방식이 조금 독특합니다. 별(*)을 붙이는 이유
    // std::min_element가 최솟값 숫자 자체를 반환하는 게 아니라, 최솟값이 위치한 주소값을 반환하기 위함입니다.
    // 별(*)을 붙여 역참조를 해서 주소에 있는 알맹이 값을 가져옵니다.
    double min = *std::min_element(latencies.begin(), latencies.end());
    double max = *std::max_element(latencies.begin(), latencies.end());

    std::cout << "================ [5. C++ Native TRT Video 벤치마크] ================" << std::endl;
    std::cout << "처리 프레임 수: " << frameCount << " / " << totalFrames << std::endl;
    std::cout << "Avg Latency: " << cv::format("%.2f", avg) << " ms" << std::endl;
    std::cout << "Min Latency: " << cv::format("%.2f", min) << " ms" << std::endl;
    std::cout << "Max Latency: " << cv::format("%.2f", max) << " ms" << std::endl;
    std::cout << "Avg FPS: " << cv::format("%.1f", 1000.0 / avg) << " FPS" << std::endl;
    std::cout << "=========================================================================" << std::endl;

    return 0;
}
```
