# YOLOv8 환경별 추론 속도 비교 (PyTorch, ONNX Runtime, TensorRT)

이 문서는 YOLOv8 Nano (`yolov8n`) 모델의 전처리 및 후처리 과정을 제외하고, **순수 모델 추론 시간(Inference Latency)** 만을 독립적으로 측정한 결과 기록입니다. 동일한 입력 조건에서 1,000회 연속 추론을 수행하여 평균, 최소, 최대 지연 시간을 측정했습니다.

---

## 📊 1,000회 추론 속도 요약

| 실행 환경 및 프레임워크 | 메모리 옵션 | 평균 소요 시간 (Avg) | 최소 소요 시간 (Min) | 최대 소요 시간 (Max) |
| :--- | :--- | :---: | :---: | :---: |
| **1. PyTorch (.pt)** | Pure `nn.Module` (CUDA) | **8.02 ms** | 7.30 ms | 17.14 ms |
| **2. ONNX Runtime** | `CUDAExecutionProvider` | **7.52 ms** | 7.32 ms | 9.09 ms |
| **3. Python TensorRT** | Standard Host-Device | **3.36 ms** | 3.24 ms | 5.89 ms |
| **4. Python TensorRT** | Zero-Copy (Page-locked) | **3.35 ms** | 3.22 ms | 5.33 ms |
| **5. C++ TensorRT** | Pure C++ CUDA Stream | **1.68 ms** | **1.62 ms** | **2.50 ms** |

---

> **참고**: C++ 환경에서 TensorRT를 사용할 때 파이썬 인터프리터 및 바인딩 오버헤드가 제거되어 가장 뛰어난 속도와 안정적인 최소/최대 지연 시간 범위를 보여줍니다.

<summary><b>1. PyTorch (.pt) 측정 코드 및 결과</b></summary>

```python
import time
import numpy as np
import torch
from ultralytics import YOLO

pt_model = YOLO('yolov8n.pt')
model = pt_model.model.eval().cuda() # PyTorch nn.Module 변환 및 GPU 이관

dummy_input = torch.ones((1, 3, 640, 640), dtype=torch.float32).cuda()

# 예열 (Warm-up) 50회
with torch.no_grad():
    for _ in range(50):
        _ = model(dummy_input)
torch.cuda.synchronize()

latencies = []
with torch.no_grad():
    for _ in range(1000):
        start = time.time()

        _ = model(dummy_input)
        torch.cuda.synchronize() # GPU 연산 완료 대기

        latencies.append((time.time() - start) * 1000)

print('================ [1. PyTorch 1000회 벤치마크] ================')
print(f'Avg Latency: {np.mean(latencies):.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print('==============================================================')
```

```python
import time
import numpy as np

providers = ['CUDAExecutionProvider', 'CPUExecutionProvider']
```

<summary><b>TensorRT 가속 엔진 (.engine) 변환 코드</b></summary>

```python
import os
import tensorrt as trt

# 로거 및 빌더 생성
TRT_LOGGER = trt.Logger(trt.Logger.WARNING)
builder = trt.Builder(TRT_LOGGER)

# 네트워크 정의 생성
network = builder.create_network()
parser = trt.OnnxParser(network, TRT_LOGGER)

# ONNX 파일 파싱
onnx_path = 'yolov8n.onnx'
if not os.path.exists(onnx_path):
    raise FileNotFoundError(f'{onnx_path} 파일이 없습니다.')

with open(onnx_path, 'rb') as model:
    if not parser.parse(model.read()):
	print('ONNX 파싱 실패')
	raise RuntimeError('ONNX 파서 에러')

# 최적화 빌드 설정
config = builder.create_builder_config()

# 직렬화된 엔진 바이너리 생성
serialized_engine = builder.build_serialized_network(network, config)

if serialized_engine is None:
    raise RuntimeError('가속 엔진 빌드 프로세스 실패')

# .engine 파일 저장
engine_path = 'yolov8n.engine'
with open(engine_path, 'wb') as f:
    f.write(serialized_engine)

# 엔진 로드 및 실행 context 준비
runtime = trt.Runtime(TRT_LOGGER)
with open(engine_path, 'rb') as f:
    engine = runtime.deserialize_cuda_engine(f.read())

context = engine.create_execution_context()
```

<summary><b>3. Python TensorRT (기본 메모리 할당) 측정 코드 및 결과</b></summary>

```python
import time
import numpy as np
import tensorrt as trt
import pycuda.driver as cuda

# 입출력 버퍼 할당
input_shape = (1, 3, 640, 640)
input_size = trt.volume(input_shape) * np.dtype(np.float32).itemsize

output_shape = engine.get_tensor_shape(engine.get_tensor_name(1))
output_size = trt.volume(output_shape) * np.dtype(np.float32).itemsize

d_input = cuda.mem_alloc(input_size)
d_output = cuda.mem_alloc(output_size)
stream = cuda.Stream()

# 입력 데이터 생성
host_input = np.ones(input_shape, dtype=np.float32)
host_output = np.empty(output_shape, dtype=np.float32)

# 텐서 주소 연결
context.set_tensor_address(engine.get_tensor_name(0), int(d_input))
context.set_tensor_address(engine.get_tensor_name(1), int(d_output))

# 예열 (50회)
for _ in range(50):
    cuda.memcpy_htod_async(d_input, host_input, stream)
    context.execute_async_v3(stream_handle=stream.handle)
    cuda.memcpy_dtoh_async(host_output, d_output, stream)
    stream.synchronize()

# 1000회 추론 측정
latencies = []
for _ in range(1000):
    start = time.time()
    
    context.execute_async_v3(stream_handle=stream.handle)
    stream.synchronize()
    
    latencies.append((time.time() - start) * 1000)

print('================ [3. Python TensorRT (기본) 1000회 벤치마크] ================')
print(f'Avg Latency: {np.mean(latencies):.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print('=========================================================================')
```

<summary><b>4. Python TensorRT (Zero-Copy Pinned Memory) 측정 코드 및 결과</b></summary>

```python
import time
import numpy as np
import tensorrt as trt
import pycuda.driver as cuda

input_shape = (1, 3, 640, 640)
output_shape = tuple(engine.get_tensor_shape(engine.get_tensor_name(1)))

# Pinned Memory(Page-locked), Device Memory 할당
h_input = cuda.pagelocked_empty(input_shape, dtype=np.float32)
h_output = cuda.pagelocked_empty(output_shape, dtype=np.float32)
d_input = cuda.mem_alloc(h_input.nbytes)
d_output = cuda.mem_alloc(h_output.nbytes)

stream = cuda.Stream()

np.copyto(h_input, np.ones(input_shape, dtype=np.float32))

context.set_tensor_address(engine.get_tensor_name(0), int(d_input))
context.set_tensor_address(engine.get_tensor_name(1), int(d_output))

# 최초 Host to Device 전송
cuda.memcpy_htod_async(d_input, h_input, stream)
stream.synchronize()

# 50회 예열
for _ in range(50):
    context.execute_async_v3(stream_handle=stream.handle)
    stream.synchronize()

# 1000회 추론 측정
latencies = []
for _ in range(1000):
    start = time.time()

    context.execute_async_v3(stream_handle=stream.handle)
    stream.synchronize()
    latencies.append((time.time() - start) * 1000)

print('================ [4. Python TensorRT (Zero-Copy) 1000회 벤치마크] ================')
print(f'Avg Latency: {np.mean(latencies):.2f} ms')
print(f'Min Latency: {np.min(latencies):.2f} ms')
print(f'Max Latency: {np.max(latencies):.2f} ms')
print('=================================================================================')
```

<summary><b>5. C++ TensorRT 구현 코드 및 측정 결과</b></summary>

### `YoloTRT.h`
```cpp
#ifndef YOLO_TRT_H
#define YOLO_TRT_H

#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <opencv2/opencv.hpp>
#include <NvInfer.h>
#include <cuda_runtime_api.h>

class TRTLogger : public nvinfer1::ILogger {
    void log(Severity severity, const char* msg) noexcept override {
        if (severity <= Severity::kWARNING) {
            std::cout << "[TensorRT] " << msg << std::endl;
        }
    }
};

class YoloTRT {
public:
    YoloTRT(const std::string& enginePath);
    ~YoloTRT();
    void inferOnly();

private:
    TRTLogger gLogger;
    nvinfer1::IRuntime* runtime = nullptr;
    nvinfer1::ICudaEngine* engine = nullptr;
    nvinfer1::IExecutionContext* context = nullptr;

    cudaStream_t stream = nullptr;

    void* d_input = nullptr;
    void* d_output = nullptr;

    size_t inputSize = 1 * 3 * 640 * 640 * sizeof(float);
    size_t outputSize = 1 * 84 * 8400 * sizeof(float);
};
#endif // YOLO_TRT_H
```

### `YoloTRT.cpp`
```cpp
#include "YoloTRT.h"

YoloTRT::YoloTRT(const std::string& enginePath) {
    std::ifstream file(enginePath, std::ios::binary);
    if (!file.good()) {
        std::cerr << "엔진 파일을 찾을 수 없습니다: " << enginePath << std::endl;
        return; 
    }
    file.seekg(0, file.end);
    size_t size = file.tellg();
    file.seekg(0, file.beg);

    std::vector<char> engineData(size);
    file.read(engineData.data(), size);
    file.close();

    runtime = nvinfer1::createInferRuntime(gLogger);
    engine = runtime->deserializeCudaEngine(engineData.data(), size);
    context = engine->createExecutionContext();

    cudaStreamCreate(&stream);

    cudaMalloc(&d_input, inputSize);
    cudaMalloc(&d_output, outputSize);

    const char* inputName = engine->getIOTensorName(0);
    const char* outputName = engine->getIOTensorName(1);
    context->setTensorAddress(inputName, d_input);
    context->setTensorAddress(outputName, d_output);
}

YoloTRT::~YoloTRT() {
    if (stream) cudaStreamDestroy(stream);
    if (d_input) cudaFree(d_input);
    if (d_output) cudaFree(d_output);
    if (context) delete context;
    if (engine) delete engine;
    if (runtime) delete runtime;
}

void YoloTRT::inferOnly() {
    this->context->enqueueV3(this->stream);
    cudaStreamSynchronize(this->stream);
}
```

### `main.cpp`
```cpp
#include "YoloTRT.h"
#include <chrono>
#include <iostream>
#include <vector>
#include <numeric>
#include <algorithm>

int main() {
    YoloTRT detector("yolov8n.engine");

    // 예열 (GPU 클럭 세팅을 위해 50회 수행)
    for (int i = 0; i < 50; ++i) {
        detector.inferOnly();
    }

    const int iterations = 1000;
    std::vector<double> latencies;
    latencies.reserve(iterations);

    // 1000회 반복 측정
    for (int i = 0; i < iterations; ++i) {
        auto start = std::chrono::high_resolution_clock::now();
        detector.inferOnly();
        auto end = std::chrono::high_resolution_clock::now();

        double ms = std::chrono::duration<double, std::milli>(end - start).count();
        latencies.push_back(ms);
    }

    // 결과 산출
    double sum = std::accumulate(latencies.begin(), latencies.end(), 0.0);
    double avg = sum / iterations;
    double min = *std::min_element(latencies.begin(), latencies.end());
    double max = *std::max_element(latencies.begin(), latencies.end());

    std::cout << "================ [C++ 순수 TRT 1000회 벤치마크] ================" << std::endl;
    std::cout << "Avg Latency: " << cv::format("%.2f", avg) << " ms" << std::endl;
    std::cout << "Min Latency: " << cv::format("%.2f", min) << " ms" << std::endl;
    std::cout << "Max Latency: " << cv::format("%.2f", max) << " ms" << std::endl;
    std::cout << "==================================================================" << std::endl;
}
```
